8     internal static bool Validate(SimulationWorld world, string actorId, InventoryRequest request, out string reason, out int duration, out int ap,
9         out InventoryPackingPlan? packing)
10     {
11         var state = world.Inventory; duration = 0; ap = 0; packing = null;
12         reason = "Inventory is unavailable";
13         if (request is { TargetId: not null, Kind: not InventoryActionKind.PresentBadge })
14         { reason = "Only badge presentation accepts a reader target."; return false; }
15         if (state.Enabled && request?.Kind is InventoryActionKind.AutoPlace or InventoryActionKind.Arrange)
16         {
17             try { packing = InventoryPackingActions.Quote(world, actorId, request, out duration, out ap); reason = "Ready"; return true; }
18             catch (Exception ex) when (ex is ArgumentException or InvalidOperationException or KeyNotFoundException)
19             { reason = ex is ArgumentException ? ex.Message : "Unknown packing destination."; return false; }
20         }
21         if (!state.Enabled || request is null || !Enum.IsDefined(request.Kind) || !state.Items.TryGetValue(request.ItemId, out var item)) return false;
22         var access = InventoryAccessService.Placement(world, actorId, item.Placement);
23         if (!access.Allowed) { reason = access.Reason; return false; }
24         if (InventoryGround.Covering(state, item).FirstOrDefault() is { } cover)
25         { reason = "Covered by " + state.Name(cover.Id) + "; move covering items first"; return false; }
26         var definition = state.Definition(item.Id);
27         duration = definition.HandleMs + InventoryAccessService.HandlingMs(state, item.Placement);
28         ap = definition.HandleAp + AccessAp(state, item.Placement);
29         try
30         {
31             switch (request.Kind)
32             {
33                 case InventoryActionKind.PresentBadge:
34                     DoorCredentialService.ValidatePresentation(world, actorId, request);
35                     duration = DoorCredentialService.PresentMs; ap = DoorCredentialService.PresentAp;
36                     break;
37                 case InventoryActionKind.Move:
38                     if (request.Destination is not { } destination || request.RegionId is not null) throw new ArgumentException("Choose a physical destination");
39                     destination = InventoryPlacement.Destination(state, item, destination);
40                     if (InventoryBodyFit.SamePlace(state, item, destination)) throw new ArgumentException("Item is already there");
41                     InventoryWear.ValidateAction(state, item, item.Placement);
42                     var target = InventoryAccessService.Placement(world, actorId, destination);
43                     if (!target.Allowed) throw new ArgumentException(target.Reason);
44                     InventoryWear.ValidateAction(state, item, destination);
45                     duration += InventoryAccessService.HandlingMs(state, destination);
46                     ap += AccessAp(state, destination);
47                     var candidate = new InventoryState(state);
48                     candidate.Items[item.Id] = item with { Placement = destination };
49                     InventoryPlacement.Validate(candidate);
50                     break;
51                 case InventoryActionKind.Open:
52                 case InventoryActionKind.Close:
53                     if (request.RegionId is null || request.Destination is not null) throw new ArgumentException("Choose a compartment");
54                     var region = state.Region(item.Id, request.RegionId);
55                     if (region.Attachment is not null || !region.Sealable) throw new ArgumentException("This region cannot be sealed; inspect its contents directly.");
56                     var compartmentAccess = InventoryWear.Compartment(state, item.Id, region);
57                     if (!compartmentAccess.Allowed) throw new ArgumentException(compartmentAccess.Reason);
58                     if (state.IsOpen(item.Id, region.Id) == (request.Kind == InventoryActionKind.Open)) throw new ArgumentException("Compartment is already in that state");
59                     duration = Math.Max(150, region.AccessMs + InventoryAccessService.HandlingMs(state, item.Placement));
60                     ap = Math.Max(1, region.AccessAp + AccessAp(state, item.Placement));
61                     break;
62                 case InventoryActionKind.OpenGarment:
63                 case InventoryActionKind.CloseGarment:
64                     if (definition.Wear is not { OpenExposes.Length: > 0 } wear || request.RegionId is not null || request.Destination is not null)
65                         throw new ArgumentException("This garment has no supported opening");
66                     if (item.GarmentOpen == (request.Kind == InventoryActionKind.OpenGarment)) throw new ArgumentException("Garment is already in that state");
67                     InventoryWear.ValidateAction(state, item, item.Placement, wear.OpenExposes, fastening: true);
68                     var closureCandidate = new InventoryState(state);
69                     closureCandidate.Items[item.Id] = item with { GarmentOpen = request.Kind == InventoryActionKind.OpenGarment };
70                     InventoryPlacement.Validate(closureCandidate);
71                     duration = wear.FastenMs + InventoryAccessService.HandlingMs(state, item.Placement);
72                     ap = wear.FastenAp + AccessAp(state, item.Placement);
73                     break;
74                 case InventoryActionKind.PowerOn:
75                 case InventoryActionKind.PowerOff:
76                     if (definition.Capability is null || request.Destination is not null || request.RegionId is not null) throw new ArgumentException("This item has no power control");
77                     if (item.Powered == (request.Kind == InventoryActionKind.PowerOn)) throw new ArgumentException("Equipment is already in that power state");
78                     break;
79             }
80         }
81         catch (Exception ex) when (ex is ArgumentException or InvalidOperationException or KeyNotFoundException or NullReferenceException)
82         { reason = ex is ArgumentException ? ex.Message : "Invalid physical destination or compartment"; return false; }
83         reason = "Ready"; return true;

85     internal static int AccessAp(InventoryState state, ItemPlacement placement)
86     {
87         var cost = 0;
88         for (var depth = 0; depth <= InventoryHierarchy.MaxDepth; depth++)
89         {
90             cost += state.Region(placement.ParentId, placement.RegionId).AccessAp;
91             if (!state.Items.TryGetValue(placement.ParentId, out var parent)) return cost;
92             placement = parent.Placement;
93         }
94         throw new ArgumentException("Inventory hierarchy is too deep");

96     public static bool Start(SimulationWorld world, InventoryRequest request, out string reason)
97     {
98         var actor = world.Player;
99         reason = world.ManualPause || world.LocalHold ? "Inventory handling waits with the local simulation."
100             : world.Inventory.Pending is not null ? "Finish the current inventory action first."
101             : world.LocalMs < actor.StartledUntilMs ? "Recovering from startle." : "Ready";
102         if (reason != "Ready") return false;
103         if (!Validate(world, actor.Id, request, out reason, out var duration, out var ap, out var packing)) return false;
104         request = InventoryPlacement.Resolve(world.Inventory, request);
105         if (world.Combat.Active && !TacticalCombatSystem.HandleInventory(world, ap, duration, out reason)) return false;
106         MovementSystem.StopInPlace(actor);
107         world.Inventory.Pending = new(actor.Id, request, actor.Position, duration, duration, world.Inventory.Revision, ap, world.Combat.Active, Packing: packing);
108         reason = $"Handling {world.Inventory.Name(request.ItemId)}: {duration / 1000.0:0.00}s" + (world.Combat.Active ? $" · {ap} AP" : "");
109         PlayerFeedback.Add(world, MessageKind.Equipment, "Inventory: " + reason);
110         return true;

120     public static void Advance(SimulationWorld world)
121     {
122         var state = world.Inventory;
123         if (state.Pending is not { } work) return;
124         var actor = world.Actors.Single(a => a.Id == work.ActorId);
125         var abort = work.AbortReason ?? (actor.Position != work.Position ? "Character moved" : world.LocalMs < actor.StartledUntilMs ? "Startled while handling equipment" : null);
126         work = work with { RemainingMs = work.RemainingMs - SpatialSimulation.TickMs, AbortReason = abort };
127         state.Pending = work;
128         if (work.RemainingMs > 0) return;
129         state.Pending = null;
130         if (work.AbortReason is not null || work.Revision != state.Revision
131             || !Validate(world, work.ActorId, work.Request, out _, out _, out _, out var verified)
132             || !InventoryPackingActions.Same(work.Packing, verified))
133         { state.SecureQueue.Clear(); PlayerFeedback.Add(world, MessageKind.Warning, "Inventory: action failed: " + (work.AbortReason ?? "Item, access or destination changed")); return; }
134         if (work.Packing is { } plan)
135         {
136             foreach (var move in plan.Moves) state.Items[move.ItemId] = state.Items[move.ItemId] with { Placement = move.To };
137             state.Revision++;
138             world.Log("inventory", work.Request.ItemId, work.Request.Kind.ToString(), "Atomic region packing committed.");
139             PlayerFeedback.Add(world, MessageKind.Equipment, "Inventory: " + state.Name(work.Request.ItemId) + ": " + work.Request.Kind + " completed.");
140             return;
141         }
142         var request = work.Request; var item = state.Items[request.ItemId];
143         switch (request.Kind)
144         {
145             case InventoryActionKind.PresentBadge:
146                 DoorCredentialService.CompletePresentation(world, work.ActorId, request); break;
147             case InventoryActionKind.Move: state.Items[item.Id] = item with { Placement = request.Destination! }; break;
148             case InventoryActionKind.Open:

150                 var index = state.Regions.FindIndex(r => r.ItemId == item.Id && r.RegionId == request.RegionId);
151                 state.Regions[index] = state.Regions[index] with { Open = request.Kind == InventoryActionKind.Open }; break;
152             case InventoryActionKind.PowerOn:
153             case InventoryActionKind.PowerOff: state.Items[item.Id] = item with { Powered = request.Kind == InventoryActionKind.PowerOn }; break;
154             case InventoryActionKind.OpenGarment:
155             case InventoryActionKind.CloseGarment: state.Items[item.Id] = item with { GarmentOpen = request.Kind == InventoryActionKind.OpenGarment }; break;
156         }
157         state.Revision++;
158         world.Log("inventory", item.Id, request.Kind.ToString(), "Timed handling committed; one physical item graph.");
159         PlayerFeedback.Add(world, MessageKind.Equipment, "Inventory: " + state.Definition(item.Id).Name + ": " + request.Kind + " completed.");
