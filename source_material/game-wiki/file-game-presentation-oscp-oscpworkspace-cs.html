112     public bool Open(OscpDevice device)
113     {
114         var access = _sim.OscpAccess(device);
115         if (!access.Available) { _sim.TryUseOscpDevice(device); return false; }
116         if (IsOpen && Device == OscpDevice.Phone && device != OscpDevice.Phone) CompletePhoneSession(true);
117         Device = device; _frame.Device = device;
118         if (device == OscpDevice.Phone && !_sim.ManualPause)
119         { _ownsPhonePause = true; _sim.ManualPause = true; }
120         foreach (var child in _nav.GetChildren()) { _nav.RemoveChild(child); child.QueueFree(); }
121         _navigation.Clear();
122         foreach (var app in _apps.For(device))
123         {
124             var button = Button(_nav, device == OscpDevice.Phone ? app.Label : app.Symbol + "  " + app.Label, () => SelectApp(app.Id));
125             button.AddThemeFontSizeOverride("font_size", device == OscpDevice.Phone ? 11 : 14);
126             button.Alignment = HorizontalAlignment.Left; button.ToggleMode = true; _navigation.Add(app.Id, button);

149     public void Send(PersonnelTask task, Hex? destination = null)
150     {
151         var actor = _sim.Actors.Where(a => a != _sim.Player).ElementAt(_recipient.Selected);
152         var d = _sim.Operations.Definition!;
153         var target = destination ?? (task == PersonnelTask.Transport ? d.Home : task == PersonnelTask.Post ? d.Posts[_recipient.Selected % d.Posts.Length] : _sim.Player.Cell);
154         if (Device == OscpDevice.Phone && _sim.ManualPause)

164     public void Execute(string command)
165     {
166         _command.Text = "";
167         var parts = command.Trim().Split(' ', StringSplitOptions.RemoveEmptyEntries);
168         if (parts.Length == 0) return;
169         var app = _apps.Find(command.Trim(), Device);
170         if (app is not null) { SelectApp(app.Id); _receipt.Text = "Opened " + app.Label; return; }
171         if (parts.Length == 1 && parts[0].Equals("help", StringComparison.OrdinalIgnoreCase))
172         { _receipt.Text = "Open: status, team, records, mail, map. Orders: order <staff id> <observe|follow|post|recover|duty> [q r]."; return; }
173         if (parts.Length is 3 or 5 && parts[0].Equals("order", StringComparison.OrdinalIgnoreCase))
174         {
175             var index = Array.FindIndex(_sim.Actors.Where(a => a != _sim.Player).ToArray(), a => a.Id.Equals(parts[1], StringComparison.OrdinalIgnoreCase));
176             var task = parts[2].ToLowerInvariant() switch { "observe" => PersonnelTask.Observe, "follow" => PersonnelTask.Follow, "post" => PersonnelTask.Post, "recover" => PersonnelTask.Transport, "duty" => PersonnelTask.ResumeDuty, _ => (PersonnelTask)(-1) };
177             if (index >= 0 && Enum.IsDefined(task) && (parts.Length == 3 || task == PersonnelTask.Post && int.TryParse(parts[3], out _) && int.TryParse(parts[4], out _)))
178             {
179                 _recipient.Select(index); Send(task, parts.Length == 5 ? new(int.Parse(parts[3]), int.Parse(parts[4])) : null); return;
180             }
181         }
182         _receipt.Text = "Unknown command or unavailable application. Type help.";
183     }
