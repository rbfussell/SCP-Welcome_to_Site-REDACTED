3 /// <summary>Body-local axial coordinates. No relationship to terrain size, projection or movement.</summary>
4 public readonly record struct BodyHex(int Q, int R)
5 {
6     private static readonly BodyHex[] Directions = [new(1, -1), new(1, 0), new(0, 1), new(-1, 1), new(-1, 0), new(0, -1)];
7     public BodyHex Neighbor(int direction) => new(Q + Directions[direction].Q, R + Directions[direction].R);
8     public BodyHex Rotate(int sixthTurns)
9     {
10         if (sixthTurns is < 0 or > 5) throw new ArgumentOutOfRangeException(nameof(sixthTurns));
11         var cell = this;
12         for (var i = 0; i < sixthTurns; i++) cell = new(-cell.R, cell.Q + cell.R);
13         return cell;
14     }
15     public static BodyHex operator +(BodyHex a, BodyHex b) => new(a.Q + b.Q, a.R + b.R);
16 }
