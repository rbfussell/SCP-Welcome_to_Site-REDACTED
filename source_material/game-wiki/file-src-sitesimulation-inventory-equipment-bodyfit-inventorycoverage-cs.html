10     internal static PocketCoverage? Evaluate(InventoryState state, InventoryItem item, StorageRegion region)
11     {
12         var mount = InventoryWear.Mount(state, item, item.Placement);
13         var shape = InventoryBodyFit.Shape(state, item.Id);
14         if (mount is null || shape is null || region.AccessFootprint is null) return null;
15         var mouth = Project(region.AccessFootprint, InventoryBodyFit.Pose(state, item)!, mount);
16         var blockers = new List<string>(); var blocked = new HashSet<BodySurfaceCell>();
17         var fallback = false;
18         foreach (var outer in InventoryWear.Worn(state, item.Placement.ParentId)
19             .Where(other => other.Id != item.Id && state.Definition(other.Id).Wear is { CoversAccess: true } wear
20                 && InventoryWornOrder.Order(other) > InventoryWornOrder.Order(item))
21             .OrderByDescending(InventoryWornOrder.Order).ThenBy(other => other.Id, StringComparer.Ordinal))
22         {
23             var coverShape = InventoryBodyFit.Shape(state, outer.Id);
24             if (coverShape?.Coverage is not null)
25             {
26                 var cover = InventoryWornStack.Cells(state, outer, open: true).ToHashSet();
27                 var intersection = mouth.Where(cover.Contains).ToArray();
28                 if (intersection.Length == 0) continue;
29                 blocked.UnionWith(intersection);
30             }
31             else
32             {
33                 // Unmapped outer gear retains the conservative, existing region rule.
34                 if (!InventoryWear.Covered(state, outer).Intersect(region.AccessBody ?? mount.Coverage).Any()) continue;
35                 fallback = true;
36             }
37             blockers.Add(outer.Id);
38         }
39         var reason = blockers.Count == 0 ? "Access footprint is uncovered."
40             : region.Name + " is covered by " + string.Join(", ", blockers.Select(id => state.Definition(id).Name)) + "; expose it first";
41         return new(item.Id, region.Id, blockers.Count > 0, reason, mouth, blocked.OrderBy(c => c.Surface, StringComparer.Ordinal)
42             .ThenBy(c => c.Cell.R).ThenBy(c => c.Cell.Q).ToArray(), blockers.ToArray(), fallback);
43     }
44 }
