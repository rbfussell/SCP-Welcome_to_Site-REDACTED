58     public override void _Process(double delta)
59     {
60         if (_inventorySmokeQuit)
61         { GC.Collect(); GC.WaitForPendingFinalizers(); GetTree().Quit(); return; }
62         _runtime.Advance(delta);
63         RefreshCamera();
64         UpdateEncounter();
65         var local = _world.GetLocalMousePosition();
66         var ground = ReferenceProjection.Unproject(new(local.X, local.Y));
67         Hex? hover = _hud.Field.GetGlobalRect().HasPoint(GetGlobalMousePosition()) ? Hex.FromWorld(ground) : null;
68         _world.Hover = hover;
69         if (_previousHover != hover || _simulation.LocalMs % 200 == 0)
70         { _world.Preview = !_hud.Operations.ChoosingPost && hover is { } cell ? _simulation.PreviewPath(cell) : []; _previousHover = hover; }
71         _noticeSeconds = Math.Max(0, _noticeSeconds - delta);
72         var audioState = $"{(_audio.Muted ? "MUTED" : "AUDIO ON")} · {(_encounter.ReducedEffects ? "REDUCED EFFECTS" : "SUBTLE EFFECTS")}";
73         _hud.Refresh(_simulation, _world, hover, ground, _zoom,
74             _noticeSeconds > 0 ? _notice : _world.FieldView ? audioState + " · " + _simulation.LastMessage : null);
75         _world.QueueRedraw();
76         if (_capturePath is not null) { if (_inventorySmoke) { if (!_inventorySmokeRunning && ++_smokeFrame == 12) { _inventorySmokeRunning = true; RunInventorySmoke(); } } else if (_combatSmoke) CombatSmokeStep(); else if (_operationSmoke) OperationSmokeStep(); else SmokeStep(); }
77     }
78     private void UpdateEncounter()
79     {
80         var discovered = _feedback.Update(_simulation, _world.FieldView);
81         _audio.Update(_world.FieldView, _simulation.LocalProgressHeld, discovered);
82         _encounter.Visible = _world.FieldView;
83         _encounter.Pulse = _feedback.Pulse(_simulation.LocalMs);
84         _encounter.Caption = _feedback.Caption(_simulation.LocalMs);
85         _encounter.PlayerPoint = _world.GetTransform() * WorldView.Screen(_simulation.Player.Position);
86         _encounter.Bearings = _simulation.Awareness.Evidence
87             .Where(e => _simulation.LocalMs - e.LocalMs <= 15000 && !_simulation.VisibleToPlayer(e.Position)
88                 && Point2.Distance(e.Position, _simulation.Player.Position) > Math.Max(0.05, e.UncertaintyRadius))
89             .OrderByDescending(e => e.LocalMs).Take(4)
90             .Select(e => new ContactBearing(WorldView.Screen(e.Position) - WorldView.Screen(_simulation.Player.Position),
91                 $"{(e.Source == "sound" ? "sound ~" : e.Source is "voice" or "radio" ? "reported" : "last seen")} {(_simulation.LocalMs - e.LocalMs) / 1000.0:0.0}s",
92                 e.Classification == "hostile"))
93             .Concat(_simulation.Combat.Active ? _simulation.Combat.AcquiredContacts
94                 .Where(c => c.ObserverId == _simulation.Player.Id && !_simulation.VisibleToPlayer(c.Position)
95                     && _simulation.LocalMs - c.LocalMs <= 15000 && Point2.Distance(c.Position, _simulation.Player.Position) > 0.05)
96                 .Select(c => new ContactBearing(WorldView.Screen(c.Position) - WorldView.Screen(_simulation.Player.Position),
97                     $"training last seen {(_simulation.LocalMs - c.LocalMs) / 1000.0:0.0}s", true)) : [])
98             .Take(4).ToArray();
99         _encounter.QueueRedraw();
100     }

108     public override void _UnhandledInput(InputEvent e)
109     {
110         if (_hud.Inventory.IsOpen) { _hud.Inventory.HandleInput(e); return; }
111         if (_hud.Workspace.IsOpen) { _hud.Workspace.HandleInput(e); return; }
112         if (e is InputEventMouseMotion motion && _hud.Field.GetGlobalRect().HasPoint(motion.Position)
113             && Input.IsMouseButtonPressed(MouseButton.Middle)) _pan += motion.Relative;
114         if (e is InputEventMouseButton { Pressed: true } mouse && _hud.Field.GetGlobalRect().HasPoint(mouse.Position))
115         {
116             var local = _world.GetGlobalTransformWithCanvas().AffineInverse() * mouse.Position;
117             var target = ReferenceProjection.Unproject(new(local.X, local.Y));
118             if (mouse.ButtonIndex is MouseButton.WheelUp or MouseButton.WheelDown)
119             {
120                 _zoom = Math.Clamp(_zoom * (mouse.ButtonIndex == MouseButton.WheelUp ? 1.125f : 1 / 1.125f), 0.65f, 2.5f);
121                 RefreshCamera();
122                 var after = _world.GetGlobalTransformWithCanvas().AffineInverse() * mouse.Position;
123                 _pan += (after - local) * _zoom;
124             }
125             else if (_hud.Operations.ChoosingPost && mouse.ButtonIndex == MouseButton.Left) _hud.Operations.PlacePost(Hex.FromWorld(target));
126             else if (_hud.Operations.ChoosingPost && mouse.ButtonIndex == MouseButton.Right) _hud.Operations.CancelPost();
127             else if (mouse.ButtonIndex == MouseButton.Right || mouse.ButtonIndex == MouseButton.Left && mouse.CtrlPressed) _simulation.Face(target);
128             else if (mouse.ButtonIndex == MouseButton.Left)
129             {
130                 var victim = _simulation.Combat.Active && !mouse.ShiftPressed ? _world.PickTrainingOpponent(local) : null;
131                 if (victim is not null) _simulation.FireTraining(victim.Id);
132                 else _simulation.RequestMove(Hex.FromWorld(target), mouse.ShiftPressed);
133             }
134         }
135         if (e is not InputEventKey { Pressed: true, Echo: false } key) return;
136         switch (key.Keycode)
137         {
138             case Key.I: _hud.Workspace.Close(); _hud.Inventory.Toggle(); break;
139             case Key.Enter: _simulation.EndTurn(); break;
140             case Key.V: _simulation.UseTrainingEquipment(CombatActionKind.ReadyWeapon); break;
141             case Key.Y: _simulation.UseTrainingEquipment(CombatActionKind.Reload); break;
142             case Key.F2: _world.FieldView = !_world.FieldView; ResetCamera(); break;
143             case Key.F4: _world.DiagnosticSprites = !_world.DiagnosticSprites; break;
144             case Key.M: _audio.Muted = !_audio.Muted; break;
145             case Key.R: _encounter.ReducedEffects = !_encounter.ReducedEffects; break;
146             case Key.J: SpatialHud.ToggleComplacency(_simulation); break;
147             case Key.T: _hud.Operations.ToggleTerminal(); break;
148             case Key.P: _hud.Workspace.Toggle(OscpDevice.Phone); break;
149             case Key.C: _world.CollisionOverlay = !_world.CollisionOverlay; break;
150             case Key.G: _world.HexGrid = !_world.HexGrid; break;
151             case Key.B: _world.FloorGrid = !_world.FloorGrid; break;
152             case Key.L: _world.SightOverlay = !_world.SightOverlay; break;
153             case Key.W: _world.ReducedWalls = !_world.ReducedWalls; break;
154             case Key.H: _world.Cutaway = !_world.Cutaway; break;
155             case Key.E: _simulation.ToggleNearestDoor(); break;
156             case Key.O: _simulation.ToggleObserverAttention(); break;
157             case Key.K: _simulation.BlinkPlayer(); break;
158             case Key.F8: _simulation.ToggleLighting(); break;
159             case Key.F6: _simulation.LocalHold = !_simulation.LocalHold; break;
160             case Key.Space: _simulation.ManualPause = !_simulation.ManualPause; break;
161             case Key.Key1: _simulation.Speed = 1; break;
162             case Key.Key2: _simulation.Speed = 2; break;
163             case Key.Key3: _simulation.Speed = 4; break;
164             case Key.F5: Save(); break;
165             case Key.F9: Restore(); break;
166             case Key.Escape: _simulation.CancelMove(); break;
167             case Key.Home: ResetCamera(); break;
168         }
