100     private void EvaluatePlayerAwareness()
101     {
102         if (!_world.LocalActive) return;
103         FacilityCommunications.SenseAndReport(_world);
104         OscpService.Acquire(_world);
105         foreach (var body in Anomalies.Bodies)
106         {
107             // Bounded prototype communication: an audible, unobstructed nearby human report.
108             // The optional operation adds received radio records; general SCiPnet transport remains later work.
109             foreach (var reporter in Actors.Where(a => a != Player && a.ObserverKind == "human"))
110                 if (Point2.Distance(reporter.Position, Player.Position) <= 4 && Map.ClearSight(reporter.Position, Player.Position)
111                     && ObservationService.Inspect(Map, reporter, body.Cell.World, LocalMs).Detected)
112                     ReceivePerceivedContact(new(body.Id, body.Revision, body.Cell.World, LocalMs, "voice", reporter.Id));
113             if (VisibleToPlayer(body.Cell.World))
114                 ReceivePerceivedContact(new(body.Id, body.Revision, body.Cell.World, LocalMs, "direct", Player.Id));
115             if (OscpService.ReceivesRadio(_world))
116             foreach (var report in Operations.ReceivedReports.Where(r => r.SubjectId == body.Id && r.Position is not null && r.Source == "radio"))
117             {
118                 var reporter = Actors.Single(a => a.Name == report.Author);
119                 ReceivePerceivedContact(new(body.Id, report.Encounter, report.Position!.Value, report.LocalMs, "radio", reporter.Id));
120             }
121         }
122         PlayerFeedback.Sense(_world);
123     }

257     public void Tick()
258     {
259         if (ManualPause) return;
260         _world.WorldMs += TickMs;
261         RegionalIntakeSystem.Advance(_world);
262         if (LocalHold) return;
263         // Securing may purchase the next action at a tactical input boundary, never free local time.
264         InventorySecuring.TryStart(_world);
265         if (Combat.Active && Combat.Pending is null)
266         {
267             TrainingOpponentSystem.Observe(_world);
268             TrainingOpponentSystem.Choose(_world);
269         }
270         if (!_world.LocalActive) return;
271         _world.LocalMs += TickMs;
272         // Due anomaly effects precede detection/entry: an unseen lethal event is never held just for proximity.
273         Anomalies.Evaluate(_world, "local-timer");
274         EvaluatePlayerAwareness();
275         InventoryActions.Advance(_world);
276         if (Combat.Active)
277         {
278             var actor = Actors.Single(a => a.Id == Combat.CurrentActorId);
279             var before = actor.Cell;
280             if (Combat.Pending?.Kind == CombatActionKind.Move && MovementSystem.Advance(_world, actor.Id))
281                 PlayerFeedback.Add(_world, MessageKind.Warning, "Movement stopped: the way ahead is blocked.");
282             EmploymentSystem.Advance(_world);
283             foreach (var employee in Actors.Where(a => EmploymentSystem.Owns(_world, a.Id) && Combat.For(a.Id) is null))
284                 MovementSystem.Advance(_world, employee.Id);
285             Anomalies.Evaluate(_world, "tactical-step");
286             EvaluatePlayerAwareness();
287             TrainingOpponentSystem.Observe(_world);
288             OperationLifecycle.Advance(_world);
289             PlayerFeedback.Sense(_world);
290             TacticalCombatSystem.ResolveTick(_world, before, EvaluatePlayerAwareness);
291             return;
292         }
293         PersonnelOrderSystem.Advance(_world);
294         EmploymentSystem.Advance(_world);
295         Anomalies.Evaluate(_world, "orders");
296         DoorCredentialService.AdvanceAutomatic(_world);
297         if (MovementSystem.Advance(_world))
298         {
299             LastMessage = "Movement stopped: the way ahead is blocked.";
300             PlayerFeedback.Add(_world, MessageKind.Warning, LastMessage);
301         }
302         ObservedTransportSystem.Advance(_world);
303         Anomalies.Evaluate(_world, "local-step-complete");
304         EvaluatePlayerAwareness();
305         OperationLifecycle.Advance(_world);
306         PlayerFeedback.Sense(_world);
307         TacticalCombatSystem.Detect(_world);
308         if (Combat.Active) TrainingOpponentSystem.Observe(_world);
309     }
