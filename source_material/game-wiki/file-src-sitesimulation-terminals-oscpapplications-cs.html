14     {

16         if (manifest is null || manifest.Version != 1 || manifest.Applications is not { Length: > 0 and <= 32 })
17             throw new ArgumentException("Invalid OSCP application manifest.");
18         var names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
19         foreach (var app in manifest.Applications)
20         {
21             if (app is null || string.IsNullOrWhiteSpace(app.Id) || !app.Id.Contains(':') || string.IsNullOrWhiteSpace(app.Label)
22                 || string.IsNullOrWhiteSpace(app.Symbol) || string.IsNullOrWhiteSpace(app.View)
23                 || app.Profiles is not { Length: > 0 } || app.Profiles.Any(p => p is not ("workstation" or "phone"))
24                 || app.Profiles.Distinct().Count() != app.Profiles.Length || app.Aliases is null
25                 || app.Aliases.Append(app.Id).Any(alias => string.IsNullOrWhiteSpace(alias) || !names.Add(alias)))
26                 throw new ArgumentException("Invalid or duplicate OSCP application registration.");
27         }
28         All = Array.AsReadOnly(manifest.Applications);
29     }
30     public OscpApplication[] For(OscpDevice device) => All.Where(app => app.Profiles.Contains(Profile(device))).ToArray();
31     public OscpApplication? Find(string name, OscpDevice device) => For(device).SingleOrDefault(app => app.Id.Equals(name, StringComparison.OrdinalIgnoreCase)
