6 /// <summary>Persistent dressing precedence, with local primary depths derived from the current outfit.</summary>
7 public static class InventoryWornOrder
8 {
9     internal const long MaxOrder = 1_000_000_000;
10     public static long Order(InventoryItem item) => item.Placement.WornOrder ?? 0;
11     internal static ItemPlacement Destination(InventoryState state, InventoryItem item, ItemPlacement destination)
12     {
13         if (InventoryWear.Mount(state, item, destination) is null)
14         {
15             if (destination.WornOrder is not null) throw new ArgumentException("Only worn equipment has a dressing order.");
16             return destination;
17         }
18         var sameRoot = item.Placement.ParentId == destination.ParentId && InventoryWear.Mount(state, item, item.Placement) is not null;
19         var order = sameRoot ? Order(item) : InventoryWear.Worn(state, destination.ParentId).Select(Order).DefaultIfEmpty(0).Max() + 1;
20         if (order is < 1 or > MaxOrder) throw new ArgumentException("Dressing order exceeds its supported bound.");
21         if (destination.WornOrder is { } requested && requested != order)
22             throw new ArgumentException("Dressing cannot insert or reorder equipment; remove outer gear first.");
23         return destination with { WornOrder = order };
24     }
25     /// <summary>Physical surface stacks project to region labels; local bands and subpositions are never stored twice.</summary>
26     public static WornLayer[] Layers(InventoryState state, string id)
27     {
28         if (!state.Items.TryGetValue(id, out var item) || InventoryWear.Mount(state, item, item.Placement) is null) return [];
29         return Layers(state, item, InventoryWornStack.Build(state, item.Placement.ParentId));
30     }
31     internal static WornLayer[] Layers(InventoryState state, InventoryItem item, IReadOnlyList<WornCellLayer> stack)
32     {
33         var root = state.Content!.Roots.Single(r => r.Id == item.Placement.ParentId);
34         var body = state.BodyFits!.Body(root.BodyFamily!);
35         return stack.Where(c => c.ItemId == item.Id)
36             .Select(c => new WornLayer(InventoryWornStack.Region(body, c.Cell), c.Layer, c.SubLayer)).Distinct().ToArray();
37     }
38     internal static void Validate(InventoryState state)
39     {
40         foreach (var item in state.Items.Values)
41         {
42             var worn = InventoryWear.Mount(state, item, item.Placement) is not null;
43             if (worn ? item.Placement.WornOrder is not (>= 1 and <= MaxOrder) : item.Placement.WornOrder is not null)
44                 throw new ArgumentException("Invalid or missing worn dressing order.");
45         }
46         foreach (var root in state.Content!.Roots.Where(r => r.ActorId is not null))
47         {
48             var orders = InventoryWear.Worn(state, root.Id).Select(Order).ToArray();
49             if (orders.Distinct().Count() != orders.Length) throw new ArgumentException("Duplicate worn dressing order.");
50         }
51     }
