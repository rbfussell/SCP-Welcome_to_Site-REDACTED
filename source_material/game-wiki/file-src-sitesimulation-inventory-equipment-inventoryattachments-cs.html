64     /// <summary>Container-style exterior drop: choose free support at the requested orientation without moving existing children.</summary>
65     public AttachmentPreview PreviewAttachmentDrop(string itemId, string owner, int rotation = 0)
66     {
67         var fallback = new InventoryRequest(InventoryActionKind.Move, itemId);
68         if (!Inventory.Items.TryGetValue(itemId, out var incoming) || !Inventory.Items.ContainsKey(owner) || rotation is < 0 or > 5)
69             return new(new(false, "Choose known equipment and a valid attachment orientation.", 0, 0), fallback);
70         var profile = Inventory.Definition(itemId).Attachment;
71         var supports = Inventory.Definition(owner).Regions?.Where(r => r.Attachment is not null).OrderBy(r => r.Id, StringComparer.Ordinal).ToArray() ?? [];
72         if (profile is null || !supports.Any(s => s.Attachment!.Interface == profile.Interface))
73             return new(new(false, "No compatible exterior attachment surface.", 0, 0), fallback);
74         if (incoming.Placement.ParentId == owner && incoming.Placement.AttachmentPose is not null)
75             return new(new(false, "Already attached here; reposition it on the gear illustration.", 0, 0), fallback);
76         AttachmentPreview? refused = null;
77         // A contact need not sit at the authored origin: enumerate every support/contact alignment, not just origins on support.
78         foreach (var support in supports.Where(s => s.Attachment!.Interface == profile.Interface))
79         {
80             var origins = support.Attachment!.Cells.SelectMany(cell => profile.Contacts.Select(contact =>
81             {
82                 var rotated = contact.Rotate(rotation); return new BodyHex(cell.Q - rotated.Q, cell.R - rotated.R);
83             })).Distinct().OrderBy(c => Math.Max(Math.Abs(c.Q), Math.Max(Math.Abs(c.R), Math.Abs(c.Q + c.R))))
84                 .ThenBy(c => c.R).ThenBy(c => c.Q);
85             foreach (var cell in origins)
86             {
87                 var preview = PreviewAttachment(itemId, owner, support.Id, new(cell.Q, cell.R, rotation));
88                 if (preview.Handling.Allowed) return preview;
89                 refused ??= preview;
90             }
91         }
92         return refused ?? new(new(false, "No free compatible support; existing attachments stay in place.", 0, 0), fallback);
93     }
