9     public static void Sense(SimulationWorld world)
10     {
11         if (!world.LocalActive) return;
12         if (world.LocalMs >= world.Player.EyesClosedUntilMs)
13         {
14             var visible = world.Actors.Where(a => a != world.Player && Sees(world, a)).Select(a => a.Id)
15                 .Concat(world.Anomalies.Bodies.Where(b => ObservationService.Inspect(world.Map, world.Player, b.Cell.World, world.LocalMs).Detected).Select(b => b.Id)).ToHashSet();
16             foreach (var id in visible.Where(id => !world.Events.Visible.Contains(id)).Order(StringComparer.Ordinal))
17             {
18                 var actor = world.Actors.SingleOrDefault(a => a.Id == id);
19                 Add(world, MessageKind.Observation, $"{actor?.Name ?? id} comes into view.");
20             }
21             world.Events.Visible.Clear(); world.Events.Visible.UnionWith(visible);
22         }
23         foreach (var report in world.Operations.Reports)
24         {
25             if ((report.Source is "radio" or "scipnet") && !OscpService.ReceivesRadio(world)) continue;
26             if (!world.Events.Once("report:" + report.Id)) continue;
27             Add(world, report.SubjectId is null ? MessageKind.Orders : MessageKind.Observation, $"{report.Author}: {report.Text}");
28         }
29     }
