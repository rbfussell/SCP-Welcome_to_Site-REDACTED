17     public static ObserverFact Inspect(FacilityMap map, ActorState actor, Point2 target, long localMs, string profile = "any")
18     {
19         if (!ValidProfile(profile)) throw new ArgumentException("Unknown observer profile.");
20         var eligible = actor.ObserverKind is "human" or "ai" && (profile == "any" || profile == actor.ObserverKind);
21         var delta = target - actor.Position;
22         var direction = actor.Cell.Neighbor(actor.Facing).World - actor.Cell.World;
23         var dot = delta.Length < 0.05 ? 1 : (delta.X * direction.X + delta.Y * direction.Y) / (delta.Length * direction.Length);
24         var reason = !actor.Attending ? "not-attending"
25             : localMs < actor.EyesClosedUntilMs ? "eyes-closed"
26             : !map.LightsOn ? "unlit"
27             : delta.Length >= VisionRange ? "out-of-range"
28             : dot < FacingDotThreshold ? "outside-facing"
29             : !map.ClearSight(actor.Position, target) ? "occluded" : "visible";
30         return new(actor.Id, actor.ObserverKind, reason == "visible", eligible,
31             reason == "visible" && !eligible ? "visible; ineligible-kind" : reason);
32     }

35         => new(profile, actors.OrderBy(a => a.Id, StringComparer.Ordinal).Select(a => Inspect(map, a, target, localMs, profile)).ToArray());

37     public static int FacingToward(Point2 from, Point2 target, int previous)
38     {
39         var delta = target - from;

41         var dq = delta.X * 2; var dr = delta.Y * 2 - delta.X;
