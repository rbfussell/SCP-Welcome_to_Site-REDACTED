public static double DistanceToSegment(Point2 p, Point2 a, Point2 b)
    {
        var ab = b - a;
        var squared = ab.X * ab.X + ab.Y * ab.Y;
        var t = squared == 0 ? 0 : Math.Clamp(((p.X - a.X) * ab.X + (p.Y - a.Y) * ab.Y) / squared, 0, 1);
        return Point2.Distance(p, a + ab * t);
    }
