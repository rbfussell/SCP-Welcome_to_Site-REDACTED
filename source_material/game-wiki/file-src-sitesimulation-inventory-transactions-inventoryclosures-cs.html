6     public static InventoryClosureAction Toggle(InventoryState state, string itemId, string? regionId = null)
7     {
8         var item = state.DiagnosticItems.Single(i => i.Id == itemId);
9         if (regionId is not null)
10         {
11             var region = state.Region(itemId, regionId);
12             if (region.Attachment is not null || !region.Sealable) throw new ArgumentException("This region has no fastening.");
13             var open = state.IsOpen(itemId, regionId);
14             return new(open ? region.SecureVerb : region.OpenVerb,
15                 new(open ? InventoryActionKind.Close : InventoryActionKind.Open, itemId, RegionId: regionId));
16         }
17         var wear = state.Definition(itemId).Wear;
18         if (wear?.OpenExposes is not { Length: > 0 }) throw new ArgumentException("This garment has no supported fastening.");
19         return new(item.GarmentOpen ? wear.SecureVerb : wear.OpenVerb,
20             new(item.GarmentOpen ? InventoryActionKind.CloseGarment : InventoryActionKind.OpenGarment, itemId));
21     }

23     /// <summary>UI dismissal requests only. Explicit securing ignores this convenience preference.</summary>
24     public static bool SecureOnClose(InventoryState state, string itemId, string? regionId = null)
25     {
26         if (regionId is not null)
27             return state.Region(itemId, regionId) is { Attachment: null, Sealable: true } && (state.Regions.Single(r => r.ItemId == itemId && r.RegionId == regionId).SecureOnClose ?? state.Region(itemId, regionId).SecureOnClose);
28         return state.Items[itemId].SecureOnClose ?? state.Definition(itemId).Wear!.SecureOnClose;
29     }
30     public static InventoryRequest[] OnDismissal(InventoryState state, IEnumerable<InventoryRequest> candidates)
31     {
32         return candidates.Where(r => r.Kind switch
33         {
34             InventoryActionKind.Close when r.RegionId is not null && r.Destination is null => SecureOnClose(state, r.ItemId, r.RegionId),
35             InventoryActionKind.CloseGarment when r.RegionId is null && r.Destination is null
36                 => state.Definition(r.ItemId).Wear?.OpenExposes is { Length: > 0 } && SecureOnClose(state, r.ItemId),
37             _ => throw new ArgumentException("Dismissal can only propose closing a compartment or garment fastening.")
38         }).Distinct().ToArray();
