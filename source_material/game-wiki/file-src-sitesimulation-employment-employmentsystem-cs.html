11     {
12         var state = world.Employment;
13         if (state.Definition is not { } d) return;
14         if (d.Posts is null || d.Appointments is null || d.Posts.Length == 0 || d.Appointments.Length == 0
15             || d.Posts.Any(p => p is null || string.IsNullOrWhiteSpace(p.Id) || p.Id.Length > 80 || string.IsNullOrWhiteSpace(p.Name)

17             || d.Posts.Select(p => p.Id).Distinct().Count() != d.Posts.Length
18             || d.Appointments.Any(a => a is null || a.ActorId == world.Player.Id || !world.Actors.Any(actor => actor.Id == a.ActorId && actor.ObserverKind == "human")
19                 || a.SupervisorId != world.Operations.Definition?.CommanderId || !d.Posts.Any(p => p.Id == a.PostId)
20                 || string.IsNullOrWhiteSpace(a.Role) || a.Role.Length > 100 || !world.Map.Walkable(a.Station) || !world.Map.Walkable(a.Rest)
21                 || a.Rest == a.Station || a.RestFacing is < 0 or > 5 || a.CycleMs is < 1000 or > 86400000
22                 || a.StartMs < 0 || a.StartMs >= a.CycleMs || a.DurationMs <= 0 || a.DurationMs >= a.CycleMs
23                 || a.MaxOvertimeMs < 0 || a.MaxOvertimeMs > a.CycleMs - a.DurationMs
24                 || new[] { a.CycleMs, a.StartMs, a.DurationMs, a.MaxOvertimeMs }.Any(ms => ms % SpatialSimulation.TickMs != 0))
25             || d.Appointments.Select(a => a.ActorId).Distinct().Count() != d.Appointments.Length
26             || d.Appointments.Select(a => a.Station).Distinct().Count() != d.Appointments.Length)
27             throw new ArgumentException("Invalid employment definitions or appointment.");
28         state.Duties.AddRange(d.Appointments.Select(a => Scheduled(a, 0)
29             ? new StaffDuty(a.ActorId, DutyPhase.Reporting, "Reporting for assigned duty", 0)

56     public static void Advance(SimulationWorld world)
57     {
58         if (world.Employment.Definition is not { } d) return;
59         foreach (var a in d.Appointments)
60         {
61             var actor = world.Actors.Single(actor => actor.Id == a.ActorId);
62             var state = world.Employment.Duties.Single(s => s.ActorId == a.ActorId);
63             var post = d.Posts.Single(p => p.Id == a.PostId);
64             var scheduled = Scheduled(a, world.LocalMs);
65             if (world.Combat.Active && world.Combat.For(actor.Id) is not null)
66             {
67                 Set(world, a, DutyPhase.Interrupted, "Tactical participation suspends employment", state.ReliefSinceMs, state.Release);
68                 continue;
69             }
70             if (HasOrder(world, actor.Id))
71             {
72                 Set(world, a, DutyPhase.Interrupted, "Authorized order suspends employment; return when released", state.ReliefSinceMs, state.Release);
73                 continue;
74             }
75             if (world.LocalMs < actor.StartledUntilMs)
76             {
77                 MovementSystem.StopInPlace(actor);
78                 Set(world, a, DutyPhase.Blocked, "Recovering from startle; post coverage unavailable", state.ReliefSinceMs, state.Release);
79                 continue;
80             }
81             // Shift edges are processed once on a local tick; a hold never replays work or handovers.
82             var reliefSince = scheduled ? null : state.ReliefSinceMs;
83             var release = scheduled ? DutyRelease.None : state.Release;
84             if (!scheduled && release == DutyRelease.None)
85             {
86                 var wasWorking = state.Phase is not DutyPhase.OffShift && (state.Phase != DutyPhase.Reporting || Ready(world, a));
87                 if (!wasWorking && reliefSince is null) release = DutyRelease.OffShift;
88                 else
89                 {
90                     var phaseMs = (world.LocalMs % a.CycleMs - a.StartMs + a.CycleMs) % a.CycleMs;
91                     reliefSince ??= world.LocalMs - (phaseMs - a.DurationMs);
92                     var replacements = d.Appointments.Count(other => other.ActorId != actor.Id && other.PostId == a.PostId
93                         && Scheduled(other, world.LocalMs) && Ready(world, other)
94                         && world.Employment.Duties.Any(s => s.ActorId == other.ActorId && s.Phase == DutyPhase.OnDuty));
95                     release = replacements >= post.MinimumStaff ? DutyRelease.Relieved
96                         : world.LocalMs - reliefSince >= a.MaxOvertimeMs ? DutyRelease.OvertimeExpired : DutyRelease.None;
97                 }
98             }
99             if (!scheduled && release != DutyRelease.None)
100             {
101                 var reason = release switch
102                 {
103                     DutyRelease.OvertimeExpired => "Overtime limit reached; leaving even without relief",
104                     DutyRelease.Relieved => "Relieved by ready on-duty staff; off shift",
105                     _ => "Off shift; not assigned this interval"
106                 };
107                 Travel(world, a, actor, a.Rest, DutyPhase.OffShift, reason, reliefSince, release);
108                 if (!actor.Moving && actor.Cell == a.Rest) actor.Facing = a.RestFacing;
109                 continue;
110             }
111             var phase = scheduled ? DutyPhase.OnDuty : DutyPhase.AwaitingRelief;
112             var detail = scheduled ? "At assigned observation station" : "Relief absent; bounded overtime at post";
113             if (actor.Cell == a.Station && !actor.Moving && actor.StoppedStep is null)
114             {
115                 actor.Facing = ObservationService.FacingToward(actor.Position, post.LookAt, actor.Facing);
116                 if (!Ready(world, a))
117                 { Set(world, a, DutyPhase.Blocked, "Post not ready: attention, light or sight obstructed", reliefSince); continue; }
118             }
119             Travel(world, a, actor, a.Station, phase, detail, reliefSince, DutyRelease.None);
120         }
121         Publish(world);
122     }
123     private static void Travel(SimulationWorld world, StaffAppointment a, ActorState actor, Hex destination,
124         DutyPhase arrived, string detail, long? reliefSince, DutyRelease release)
125     {
126         if (actor.Cell == destination && !actor.Moving && actor.StoppedStep is null)
127         { Set(world, a, arrived, detail, reliefSince, release); return; }
128         if (actor.Moving && actor.Route[^1] != destination) MovementSystem.StopInPlace(actor);
129         // Employment only owns this route while no acknowledged command owns the employee.
130         if (!actor.Moving)
131         {
132             var path = MovementSystem.PlanRoute(world, actor, destination);
133             if (path.Count == 0)
134             { Set(world, a, DutyPhase.Blocked, release != DutyRelease.None ? "Off-shift rest route blocked; not providing coverage" : "Duty route blocked; waiting for access or clearance", reliefSince, release); return; }
135             MovementSystem.CommitRoute(actor, path, false);
136         }
137         Set(world, a, arrived == DutyPhase.OffShift ? DutyPhase.OffShift : DutyPhase.Reporting,
138             arrived == DutyPhase.OffShift ? detail + "; travelling to rest point" : "Travelling to assigned post", reliefSince, release);

140     private static void Publish(SimulationWorld world)
141     {
142         foreach (var duty in world.Employment.Duties)
143         {
144             var actor = world.Actors.Single(a => a.Id == duty.ActorId);
145             var previous = world.Employment.Reports.SingleOrDefault(r => r.ActorId == duty.ActorId);
146             if (previous is not null && previous.Phase == duty.Phase && previous.Reason == duty.Reason) continue;
147             var voice = FacilityCommunications.Voice(world, actor, world.Player);
148             if (!voice && !(OscpService.ReceivesRadio(world) && FacilityCommunications.Radio(world, actor.Id))) continue;
149             world.Employment.Reports.RemoveAll(r => r.ActorId == actor.Id);
150             world.Employment.Reports.Add(new(actor.Id, duty.Phase, duty.Reason, world.LocalMs));
151             world.Operations.Report(world, actor.Name, $"Employment: {duty.Phase} · {duty.Reason}", source: voice ? "voice" : "radio");
152         }
