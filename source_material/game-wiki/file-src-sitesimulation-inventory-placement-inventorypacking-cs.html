10     public static InventoryPackingPlan Plan(InventoryState state, string owner, string regionId, string? incoming)
11     {
12         var region = state.Region(owner, regionId);
13         if (region.MountSlot is not null || region.Attachment is not null) throw new ArgumentException("Arrange applies to storage, not body or attachment mounts.");
14         var ids = state.Children(owner, regionId).Select(i => i.Id).Where(id => id != incoming).ToList();
15         if (incoming is not null) ids.Add(incoming);
16         var shapes = ids.Select(id =>
17         {
18             var definition = state.Definition(id);
19             if (definition.ThicknessMm > region.HeightMm) throw new ArgumentException("Item is too thick for this compartment.");
20             if (region.AcceptTags is { Length: > 0 } tags && !definition.Tags.Any(tags.Contains)) throw new ArgumentException("Compartment rejects this item category.");
21             var size = InventoryPlacement.Footprint(definition, region, false);
22             return new Shape(id, size.Width, size.Height, state.Items[id].Placement.Rotated);
23         }).ToArray();
24         // Moving a child out of a box already in this region must not count it twice.
25         var massState = state;
26         if (incoming is not null)
27         {
28             massState = new InventoryState(state);
29             massState.Items[incoming] = massState.Items[incoming] with { Placement = new(owner, regionId) };
30         }
31         if (ids.Sum(id => InventoryHierarchy.Mass(massState, id)) > region.MaxMassGrams) throw new ArgumentException("Compartment load limit exceeded.");
32         var available = region.Mask?.Length ?? region.Columns * region.Rows;
33         if (shapes.Sum(s => s.Width * s.Height) > available) throw new ArgumentException("Container full: insufficient usable cells; rearranging cannot help.");
34         var cells = new bool[region.Columns * region.Rows];
35         for (var i = 0; i < cells.Length; i++) cells[i] = region.Mask is not null && !region.Mask.Contains(i);
36         bool Fits(bool[] occupied, int x, int y, int width, int height)
37         {
38             if (x + width > region.Columns || y + height > region.Rows) return false;
39             for (var row = y; row < y + height; row++)
40             for (var col = x; col < x + width; col++) if (occupied[row * region.Columns + col]) return false;
41             return true;
42         }
43         void Mark(Spot spot, bool occupied)
44         {
45             for (var y = spot.Y; y < spot.Y + spot.Height; y++)
46             for (var x = spot.X; x < spot.X + spot.Width; x++) cells[y * region.Columns + x] = occupied;
47         }
48         var examined = 0;
49         IEnumerable<Spot> Positions(Shape shape)
50         {
51             var choices = new List<Spot>();
52             foreach (var rotated in new[] { shape.PreferredRotation, !shape.PreferredRotation }.Distinct())
53             {
54                 if (shape.Width == shape.Height && rotated != shape.PreferredRotation) continue;
55                 var w = rotated ? shape.Height : shape.Width; var h = rotated ? shape.Width : shape.Height;
56                 for (var y = 0; y <= region.Rows - h; y++)
57                 for (var x = 0; x <= region.Columns - w; x++)
58                 {
59                     if (++examined > 200000) throw new ArgumentException("Packing search limit reached; current contents are unchanged. Try arranging a smaller selection.");
60                     if (Fits(cells, x, y, w, h))
61                     {
62                         // Compact into the upper-left, preferring contact instead of isolated holes.
63                         var contact = (x == 0 ? h : 0) + (y == 0 ? w : 0);
64                         for (var row = y; row < y + h; row++) if (x > 0 && cells[row * region.Columns + x - 1]) contact++;
65                         for (var col = x; col < x + w; col++) if (y > 0 && cells[(y - 1) * region.Columns + col]) contact++;
66                         choices.Add(new(x, y, rotated, w, h, (y + h) * 100000 + (x + w) * 1000 + y * 32 + x - contact * 5));
67                     }
68                 }
69             }
70             return choices.OrderBy(p => p.Score).ThenBy(p => p.Rotated == shape.PreferredRotation ? 0 : 1);
71         }
72         var placements = new Dictionary<string, ItemPlacement>(StringComparer.Ordinal);
73         if (incoming is not null)
74         {
75             foreach (var id in ids.Where(id => id != incoming))
76             {
77                 var item = state.Items[id]; var size = InventoryPlacement.Footprint(state.Definition(id), region, item.Placement.Rotated);
78                 Mark(new(item.Placement.X, item.Placement.Y, item.Placement.Rotated, size.Width, size.Height, 0), true);
79             }
80             var direct = Positions(shapes.Single(s => s.Id == incoming)).FirstOrDefault();
81             if (direct is not null) placements[incoming] = new(owner, regionId, direct.X, direct.Y, direct.Rotated);
82             if (placements.Count != 0) return Build(false);
83             for (var i = 0; i < cells.Length; i++) cells[i] = region.Mask is not null && !region.Mask.Contains(i);
84         }
85         var ordered = shapes.OrderByDescending(s => s.Width * s.Height).ThenByDescending(s => Math.Max(s.Width, s.Height))
86             .ThenBy(s => state.Definition(s.Id).Id, StringComparer.Ordinal).ThenBy(s => s.Id, StringComparer.Ordinal).ToArray();
87         var nodes = 0; var exhausted = false;
88         bool Search(int index)
89         {
90             if (index == ordered.Length) return true;
91             if (++nodes > SearchBudget) { exhausted = true; return false; }
92             var shape = ordered[index];
93             foreach (var spot in Positions(shape))
94             {
95                 Mark(spot, true); placements[shape.Id] = new(owner, regionId, spot.X, spot.Y, spot.Rotated);
96                 if (Search(index + 1)) return true;
97                 placements.Remove(shape.Id); Mark(spot, false);
98                 if (exhausted) return false;
99             }
100             return false;
101         }
102         if (!Search(0)) throw new ArgumentException(exhausted
103             ? "Packing search limit reached; current contents are unchanged. Try arranging a smaller selection."
104             : "Container full: its shapes cannot fit, even after arranging.");
105         return Build(true);

107         InventoryPackingPlan Build(bool repacked)
108         {
109             var moves = placements.Where(p => p.Value != state.Items[p.Key].Placement).OrderBy(p => p.Key, StringComparer.Ordinal)
110                 .Select(p => new InventoryPackingMove(p.Key, state.Items[p.Key].Placement, p.Value)).ToArray();
111             var candidate = new InventoryState(state);
112             foreach (var move in moves) candidate.Items[move.ItemId] = candidate.Items[move.ItemId] with { Placement = move.To };
113             InventoryPlacement.Validate(candidate); // Also checks root load, hierarchy/cycles and all other regions.
114             if (moves.Length == 0) throw new ArgumentException("Already arranged; no handling needed.");
115             return new(moves, repacked);
116         }
