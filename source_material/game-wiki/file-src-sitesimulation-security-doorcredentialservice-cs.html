32     {
33         handBadge = null; reason = "Access accepted.";
34         if (door.Reader is not { } reader || world.Map.OpenDoors.Contains(door.Id)) return true;
35         if (!reader.Powered) { reason = "Door reader has no power; access denied."; return false; }
36         if (!Reachable(world, actor, door)) { reason = "Approach the reader with clear physical access."; return false; }

38         var candidates = world.Inventory.Items.Values.Where(i => Valid(world.Inventory, actor, i, reader)).OrderBy(i => i.Id, StringComparer.Ordinal).ToArray();

69     {
70         if (!world.LocalActive || world.Combat.Active || world.Inventory.Pending is not null) return;
71         foreach (var actor in world.Actors.Where(a => a.Moving && world.LocalMs >= a.StartledUntilMs))
72         foreach (var door in world.Map.Definition.Walls.Where(w => w.Reader is not null && !world.Map.OpenDoors.Contains(w.Id)))
73         {
74             if (!Reachable(world, actor, door) || !Authorize(world, actor, door, out var hand, out _) || hand is not null) continue;
