3 /// <summary>Deterministic training decisions use direct observations and retained points, never hidden live targets.</summary>
4 internal static class TrainingOpponentSystem

6     public static void Observe(SimulationWorld world)
7     {
8         var state = world.Combat;
9         foreach (var c in state.Participants.Where(c => c.Points > 0))
10         {
11             var actor = world.Actors.Single(a => a.Id == c.ActorId);
12             foreach (var target in state.Participants.Where(t => t.Side != c.Side && t.Points > 0))
13             {
14                 var other = world.Actors.Single(a => a.Id == target.ActorId);
15                 if (!ObservationService.Inspect(world.Map, actor, other.Position, world.LocalMs).Detected) continue;
16                 state.Contacts.RemoveAll(p => p.ObserverId == actor.Id && p.TargetId == other.Id);
17                 state.Contacts.Add(new(actor.Id, other.Id, other.Position, world.LocalMs));
18             }
19         }
20     }
21     public static void Choose(SimulationWorld world)
22     {
23         var state = world.Combat;
24         if (!state.Active || state.Pending is not null || state.CurrentActorId == world.Player.Id) return;
25         var actor = world.Actors.Single(a => a.Id == state.CurrentActorId);
26         var c = state.For(actor.Id)!;
27         if (world.LocalMs < actor.StartledUntilMs || c.Ap == 0) { TacticalCombatSystem.EndTurn(world, actor.Id, out _); return; }
28         var target = state.Contacts.Where(p => p.ObserverId == actor.Id && state.For(p.TargetId)!.Points > 0)
29             .OrderBy(p => Point2.Distance(actor.Position, p.Position)).ThenBy(p => p.TargetId, StringComparer.Ordinal).FirstOrDefault();
30         if (target is not null)
31         {
32             if (!c.WeaponReady && TacticalCombatSystem.Equipment(world, actor.Id, CombatActionKind.ReadyWeapon, out _)) return;
33             if (c.Ammo == 0 && TacticalCombatSystem.Equipment(world, actor.Id, CombatActionKind.Reload, out _)) return;
34             // Fire revalidates actual direct sight. Retained contact is only a travel/facing cue.
35             if (TacticalCombatSystem.Fire(world, actor.Id, target.TargetId, out _)) return;
36             if (actor.Cell != Hex.FromWorld(target.Position))
37             {
38                 var route = MovementSystem.PlanRoute(world, actor, Hex.FromWorld(target.Position));
39                 // A visible target occupies its cell: approach a reachable neighbour instead.
40                 if (route.Count == 0) route = Enumerable.Range(0, 6).Select(Hex.FromWorld(target.Position).Neighbor)
41                     .Select(cell => MovementSystem.PlanRoute(world, actor, cell)).Where(p => p.Count > 0).OrderBy(p => p.Count).FirstOrDefault() ?? [];
42                 if (route.Count > 0 && TacticalCombatSystem.Move(world, actor.Id, route[Math.Min(Math.Min(3, c.Ap), route.Count) - 1], false, out _)) return;
43             }
44         }
45         if (TacticalCombatSystem.Face(world, actor.Id, actor.Cell.Neighbor((actor.Facing + 1) % 6).World, out _)) return;
46         TacticalCombatSystem.EndTurn(world, actor.Id, out _);
47     }
