10     public AnomalySystem(IEnumerable<AnomalyDefinition> definitions)
11     {
12         foreach (var definition in definitions.OrderBy(d => d.Id, StringComparer.Ordinal))
13         {
14             IAnomalyModule module = definition.Module switch
15             {
16                 "scp:650" => new Scp650Module(definition),

18             };
19             if (_bodies.Any(b => b.Id == definition.Id)) throw new ArgumentException("Duplicate anomaly identity.");
20             _bodies.Add(new(definition.Id, definition.Module, new(definition.Q, definition.R), poseCount: 6, maxRelocationHexes: 3, observerProfile: definition.ObserverProfile));
21             _modules.Add(module);
22         }
23     }

30         }
31     }

37     }

40     internal bool TrySetFlag(SimulationWorld world, string instanceId, string flag, string actorId, bool enabled, out string reason)
41     {
42         var module = _modules.SingleOrDefault(m => m.InstanceId == instanceId);
43         var actor = world.Actors.SingleOrDefault(a => a.Id == actorId);
44         reason = !world.LocalActive ? "local-held" : module is null ? "unknown-instance" : actor is null ? "unknown-actor" : "accepted";

49     internal void Restore(AnomalyBodySnapshot[] bodies, ModuleSnapshot[] modules, SimulationWorld world)
50     {
51         if (bodies is null || modules is null || bodies.Any(b => b is null) || modules.Any(m => m is null)
52             || !bodies.Select(b => b.Id).Order().SequenceEqual(_bodies.Select(b => b.Id).Order())
53             || !modules.Select(m => m.InstanceId).Order().SequenceEqual(_modules.Select(m => m.InstanceId).Order()))
54             throw new ArgumentException("Invalid anomaly instance set.");
55         world.Map.DynamicBlockers.Clear();
56         foreach (var snapshot in bodies)
57         {
58             var body = _bodies.Single(b => b.Id == snapshot.Id);
59             if (body.ModuleId != snapshot.ModuleId || (snapshot.Pose < 0 || snapshot.Pose >= body.PoseCount) || snapshot.Revision < 0)
60                 throw new ArgumentException("Invalid anomaly body state.");
61             body.Cell = snapshot.Cell; body.Pose = snapshot.Pose; body.Revision = snapshot.Revision;
62         }
63         RegisterBlockers(world.Map);
64         foreach (var module in _modules) module.Restore(modules.Single(m => m.InstanceId == module.InstanceId), world.LocalMs,
65             world.Actors.ToDictionary(a => a.Id, a => a.ObserverKind));
66         foreach (var module in _modules)
67         {
68             var body = _bodies.Single(b => b.Id == module.InstanceId);
69             var diagnostics = module.Inspect(new AnomalyWorldAccess(world, body));
70             if (diagnostics.Relocations != body.Revision || diagnostics.Observation.Observers.Length != world.Actors.Count)
71                 throw new ArgumentException("Module and spatial body history disagree.");
72         }
73     }
