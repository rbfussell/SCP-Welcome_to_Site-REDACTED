22 /// <summary>Serializable operation facts; orders, transport and terminal projections have separate services.</summary>
23 public sealed class OperationState
24 {
25     public OperationDefinition? Definition { get; }
26     internal List<PersonnelOrder> Orders { get; } = [];
27     internal List<StaffContact> Contacts { get; } = [];
28     internal List<OperationReport> Reports { get; } = [];
29     public IReadOnlyList<PersonnelOrder> OrderHistory => Orders.AsReadOnly();
30     public IReadOnlyList<OperationReport> ReceivedReports => Reports.AsReadOnly();
31     internal bool RegionalIntakeSent { get; set; }
32     public FieldPhoneState? Phone { get; set; }
33     internal HashSet<long> AcquiredReports { get; } = [];
34     public List<RadioState> Radios { get; } = [];
35     public bool TerminalPowered { get; set; } = true;
36     public bool NetworkOnline { get; set; } = true;
37     public TrolleyState Trolley { get; internal set; } = new(new(0, 0));
38     public int StableMs { get; internal set; }
39     public bool HadBreach { get; internal set; }
40     public bool Completed { get; internal set; }
41     public string PhysicalStatus { get; internal set; } = "Awaiting coverage";
42     internal long NextId { get; set; } = 1;
43     internal OperationState(string? json)
44     {
45         if (json is null) return;
46         Definition = JsonSerializer.Deserialize<OperationDefinition>(json, new JsonSerializerOptions { PropertyNameCaseInsensitive = true })
47             ?? throw new ArgumentException("Empty operation definition.");
48         Phone = new(Definition.CommanderId, Definition.FieldPhoneIssued);
49         Trolley = new(Definition.Trolley.World);
50         Radios.AddRange(Definition.RadioActors.Select(id => new RadioState(id)));
51     }
52     internal void Report(SimulationWorld world, string author, string text, Point2? position = null,
53         string? subjectId = null, long encounter = 0, string source = "system")
54     {
55         Reports.Add(new(NextId++, world.LocalMs, author, text, position, subjectId, encounter, source));
56         if (Reports.Count > 40) { AcquiredReports.Remove(Reports[0].Id); Reports.RemoveAt(0); }
57         world.Log("communications", author, "delivered", text);
58     }
59     internal OperationSnapshot Capture() => new(NextId, Orders.ToArray(), Contacts.ToArray(), Reports.ToArray(), Radios.ToArray(),
60         TerminalPowered, NetworkOnline, Trolley, StableMs, HadBreach, Completed, PhysicalStatus, Phone, AcquiredReports.Order().ToArray(), RegionalIntakeSent);
61 }
