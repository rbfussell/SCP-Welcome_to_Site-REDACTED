32     private static void Validate(PaperdollManifest m)
33     {
34         static bool Rect(float[]? r) => r is { Length: 4 } && r.All(float.IsFinite) && r[0] >= 0 && r[1] >= 0
35             && r[2] > 0 && r[3] > 0 && r[0] + r[2] <= 1.001f && r[1] + r[3] <= 1.001f;
36         if (m.Version != 2 || m.Actor != "avatar" || m.BodyFamily != "human-nine-shoulder"
37             || m.CanvasSize is not { Length: 2 } || m.CanvasSize.Any(n => n <= 0) || m.Images is not { Count: > 0 }
38             || m.Images.Any(i => System.IO.Path.GetFileName(i.Key) != i.Key || !i.Key.EndsWith(".png", StringComparison.Ordinal)
39                 || i.Value is not { Length: 2 } || i.Value.Any(n => n <= 0))
40             || m.Frames is not { Length: > 0 and <= 128 } || m.Frames.Any(f => f.Column is < 0 or > 127)
41             || m.Frames.Select(f => f.Column).Distinct().Count() != m.Frames.Length || !m.Frames.Any(f => f.Column == 0)
42             || m.Frames.Any(f => !m.Images.ContainsKey(f.File) || f.Parts is not { Length: > 0 and <= 8 }
43                 || f.Parts.Any(p => p.Side is not (null or "front" or "back") || !Rect(p.Destination) || p.BackDestination is { } back && !Rect(back)
44                     || new[] { p.Front, p.Back }.Any(r => r is not { Length: 4 } || r[0] < 0 || r[1] < 0 || r[2] <= 0 || r[3] <= 0
45                         || (long)r[0] + r[2] > m.Images[f.File][0] || (long)r[1] + r[3] > m.Images[f.File][1])))
46             || m.Frames.Single(f => f.Column == 0).Parts.Length != 1 || m.Frames.Single(f => f.Column == 0).Parts[0].Side is not null
47             || m.HeadCutouts is null || !new[] { "front", "back" }.All(side => m.HeadCutouts.TryGetValue(side, out var polygon)
48                 && polygon.Length >= 3 && polygon.All(p => p is { Length: 2 } && p.All(n => float.IsFinite(n) && n is >= 0 and <= 1)))
49             || m.Mosaic is null || !Rect(m.Mosaic.Front) || !Rect(m.Mosaic.Back) || m.Mosaic.BlockSize is < 8 or > 40
50             || m.Bindings is not { Length: > 0 and <= 128 } || m.Bindings.Select(b => b.Definition).Distinct().Count() != m.Bindings.Length
51             || m.Bindings.Any(b => string.IsNullOrWhiteSpace(b.Definition) || b.Column <= 0 || !m.Frames.Any(f => f.Column == b.Column)
52                 || (b.OpenColumn is { } open && (open <= 0 || !m.Frames.Any(f => f.Column == open)))
53                 || b.BaseCutouts is { } cuts && cuts.Any(r => !Rect(r))
54                 || b.MountColumns is { } mounts && (mounts.Count == 0 || mounts.Any(pair => string.IsNullOrWhiteSpace(pair.Key)
55                     || pair.Value <= 0 || !m.Frames.Any(f => f.Column == pair.Value)))))
56             throw new InvalidOperationException("Invalid bounded full-body paperdoll manifest.");
57     }
58     internal PaperdollLayer[] Layers(SpatialSimulation sim)
59     {
60         var root = sim.Inventory.Content!.Roots.Single(r => r.ActorId == sim.Player.Id);
61         if (root.ActorId != _manifest.Actor || root.BodyFamily != _manifest.BodyFamily) return [];
62         var result = new List<PaperdollLayer> { new(null, 0) };
63         foreach (var item in sim.BodyFitItems.Where(i => i.Placement.ParentId == root.Id
64             && sim.Inventory.Region(i.Placement.ParentId, i.Placement.RegionId).MountSlot is { } slot
65             && InventoryWear.Accepts(sim.Inventory.Definition(i.Id), slot)).OrderBy(i => i.Placement.WornOrder))
66         {
67             var binding = _manifest.Bindings.SingleOrDefault(b => b.Definition == sim.Inventory.Definition(item.Id).Id);
68             var shape = InventoryBodyFit.Shape(sim.Inventory, item.Id);
69             if (binding is null || shape is null || InventoryBodyFit.Pose(sim.Inventory, item) != shape.DefaultPose) continue;
70             var column = item.GarmentOpen ? binding.OpenColumn ?? binding.Column : binding.Column;
71             if (binding.MountColumns is { } mounts)
72             {
73                 var slot = sim.Inventory.Region(item.Placement.ParentId, item.Placement.RegionId).MountSlot!;
74                 if (!mounts.TryGetValue(slot, out column)) continue; // Unqualified mounts stay explicitly omitted.
75             }
76             result.Add(new(item.Id, column));
77         }
78         return result.ToArray();
79     }

94     internal Rect2 Bounds(Vector2 size)
95     {
96         var source = new Vector2(_manifest.CanvasSize[0], _manifest.CanvasSize[1]);
97         var scaled = source * Math.Min(size.X / source.X, size.Y / source.Y); return new((size - scaled) / 2, scaled);
98     }
99     internal bool Opaque(PaperdollLayer layer, string side, Vector2 uv)
100     {
101         foreach (var part in Parts(layer, side))
102         {
103             var dest = Destination(part, side); var local = (uv - dest.Position) / dest.Size;
104             if (local.X is < 0 or >= 1 || local.Y is < 0 or >= 1) continue;
105             var source = Source(part, side); var pixel = source.Position + local * source.Size;
106             if (_images[Frame(layer).File].GetPixel((int)pixel.X, (int)pixel.Y).A > .1f) return true;
107         }
108         return false;
109     }
110     private Vector2[] Head(string side) => _manifest.HeadCutouts[side].Select(p => new Vector2(p[0], p[1])).ToArray();
111     internal bool HeadOpaque(string side, Vector2 uv) => Geometry2D.IsPointInPolygon(uv, Head(side)) && Opaque(new(null, 0), side, uv);
112     internal string? Pick(SpatialSimulation sim, string side, Vector2 size, Vector2 position)
113     {
114         var bounds = Bounds(size); var uv = (position - bounds.Position) / bounds.Size;
115         if (HeadOpaque(side, uv)) return null;
116         return Layers(sim).Reverse().FirstOrDefault(l => Opaque(l, side, uv))?.ItemId;
117     }
118     /// <summary>Authored stand-in body cutouts prevent bare arms clipping through opaque long sleeves. No physical surface is changed.</summary>
119     private Rect2[] BaseRegions(PaperdollLayer[] layers)
120     {
121         var columns = layers.Where(l => l.ItemId is not null).Select(l => l.Column).ToHashSet();
122         var cuts = _manifest.Bindings.Where(b => columns.Contains(b.Column) || b.OpenColumn is { } open && columns.Contains(open))
123             .SelectMany(b => b.BaseCutouts ?? []).Select(r => new Rect2(r[0], r[1], r[2], r[3])).Distinct().ToArray();
124         if (cuts.Length == 0) return [new(0, 0, 1, 1)];
125         var xs = cuts.SelectMany(r => new[] { r.Position.X, r.End.X }).Concat(new float[] { 0, 1 }).Distinct().Order().ToArray();
126         var ys = cuts.SelectMany(r => new[] { r.Position.Y, r.End.Y }).Concat(new float[] { 0, 1 }).Distinct().Order().ToArray();
127         var result = new List<Rect2>();
128         for (var x = 0; x < xs.Length - 1; x++)
129         for (var y = 0; y < ys.Length - 1; y++)
130         {
131             var r = new Rect2(xs[x], ys[y], xs[x + 1] - xs[x], ys[y + 1] - ys[y]);
132             if (!cuts.Any(c => c.HasPoint(r.GetCenter()))) result.Add(r);
133         }
134         return result.ToArray();
135     }
136     /// <summary>Visual exposure only; neither this overlay nor garment alpha changes physical access or item identity.</summary>
137     internal bool MosaicVisible(SpatialSimulation sim, string side, Vector2 uv)
138         => PaperdollMosaic.Region(_manifest.Mosaic, side).HasPoint(uv) && Layers(sim) is { Length: > 0 } layers
139             && !layers.Any(l => l.ItemId is not null && Opaque(l, side, uv));
140     private void DrawMosaic(Control canvas, Rect2 bounds, string side)
141     {
142         var body = Frame(new(null, 0)); var part = body.Parts.Single();
143         var source = Source(part, side); var dest = Destination(part, side);
144         foreach (var tile in PaperdollMosaic.Tiles(_manifest.Mosaic, side, new(_manifest.CanvasSize[0], _manifest.CanvasSize[1])))
145         {
146             var sample = source.Position + (tile.GetCenter() - dest.Position) / dest.Size * source.Size;
147             var color = _images[body.File].GetPixel((int)sample.X, (int)sample.Y);
148             // Opaque coarse blocks are synthesized by Godot; no mosaic is stored in the source image.
149             color = new(MathF.Round(color.R * 12) / 12, MathF.Round(color.G * 12) / 12, MathF.Round(color.B * 12) / 12, 1);
150             canvas.DrawRect(new(bounds.Position + tile.Position * bounds.Size, tile.Size * bounds.Size), color);
151         }
152     }
153     internal void Draw(Control canvas, SpatialSimulation sim, string side, string? hover, string? selected)
154     {
155         var layers = Layers(sim); if (layers.Length == 0) return;
156         var bounds = Bounds(canvas.Size);
157         foreach (var layer in layers)
158         {
159             var tint = layer.ItemId is not null && layer.ItemId == hover ? new Color(1.3f, 1.5f, 1.25f)
160                 : layer.ItemId is not null && layer.ItemId == selected ? new Color(1.15f, 1.25f, 1.1f) : Colors.White;
161             foreach (var part in Parts(layer, side))
162             {
163                 var dest = Destination(part, side); var sourcePart = Source(part, side);
164                 foreach (var clip in layer.ItemId is null ? BaseRegions(layers) : new[] { dest })
165                 {
166                     var visible = dest.Intersection(clip); if (!visible.HasArea()) continue;
167                     var sourceClip = new Rect2(sourcePart.Position + (visible.Position - dest.Position) / dest.Size * sourcePart.Size,
168                         visible.Size / dest.Size * sourcePart.Size);
169                     canvas.DrawTextureRectRegion(_textures[Frame(layer).File], new(bounds.Position + visible.Position * bounds.Size, visible.Size * bounds.Size), sourceClip, tint);
170                 }
171             }
172             if (layer.ItemId is null) DrawMosaic(canvas, bounds, side);
173         }
174         var body = new PaperdollLayer(null, 0); var bodyPart = Frame(body).Parts.Single();
175         var source = Source(bodyPart, side); var destBody = Destination(bodyPart, side); var head = Head(side);
176         var texture = _textures[Frame(body).File];
177         canvas.DrawPolygon(head.Select(uv => bounds.Position + uv * bounds.Size).ToArray(), new[] { Colors.White },
178             head.Select(uv => (source.Position + (uv - destBody.Position) / destBody.Size * source.Size) / texture.GetSize()).ToArray(), texture);
179     }
