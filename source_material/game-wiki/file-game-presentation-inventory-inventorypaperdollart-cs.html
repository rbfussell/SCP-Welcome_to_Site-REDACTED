36     private static void Validate(PaperdollManifest m)
37     {
38         static bool Rect(float[]? r) => r is { Length: 4 } && r.All(float.IsFinite) && r[0] >= 0 && r[1] >= 0
39             && r[2] > 0 && r[3] > 0 && r[0] + r[2] <= 1.001f && r[1] + r[3] <= 1.001f;
40         if (m.Version != 2 || m.Actor != "avatar" || m.BodyFamily != "human-nine-shoulder"
41             || m.CanvasSize is not { Length: 2 } || m.CanvasSize.Any(n => n <= 0) || m.Images is not { Count: > 0 }
42             || m.Images.Any(i => !AssetFile(i.Key)
43                 || i.Value is not { Length: 2 } || i.Value.Any(n => n <= 0))
44             || m.Frames is not { Length: > 0 and <= 128 } || m.Frames.Any(f => f.Column is < 0 or > 127)
45             || m.Frames.Select(f => f.Column).Distinct().Count() != m.Frames.Length || !m.Frames.Any(f => f.Column == 0)
46             || m.Frames.Any(f => !m.Images.ContainsKey(f.File) || f.Parts is not { Length: > 0 and <= 8 }
47                 || f.Parts.Any(p => p.Side is not (null or "front" or "back") || !Rect(p.Destination) || p.BackDestination is { } back && !Rect(back)
48                     || !PaperdollWornMask.Valid(p.FrontWornCutouts) || !PaperdollWornMask.Valid(p.BackWornCutouts)
49                     || new[] { p.Front, p.Back }.Any(r => r is not { Length: 4 } || r[0] < 0 || r[1] < 0 || r[2] <= 0 || r[3] <= 0
50                         || (long)r[0] + r[2] > m.Images[f.File][0] || (long)r[1] + r[3] > m.Images[f.File][1])))
51             || m.Frames.Single(f => f.Column == 0).Parts.Length != 1 || m.Frames.Single(f => f.Column == 0).Parts[0].Side is not null
52             || m.Frames.Single(f => f.Column == 0).Parts.Any(p => p.FrontWornCutouts is not null || p.BackWornCutouts is not null)
53             || m.HeadCutouts is null || !new[] { "front", "back" }.All(side => m.HeadCutouts.TryGetValue(side, out var polygon)
54                 && polygon.Length >= 3 && polygon.All(p => p is { Length: 2 } && p.All(n => float.IsFinite(n) && n is >= 0 and <= 1)))
55             || m.Mosaic is null || !Rect(m.Mosaic.Front) || !Rect(m.Mosaic.Back) || m.Mosaic.BlockSize is < 8 or > 40
56             || m.Bindings is not { Length: > 0 and <= 128 } || m.Bindings.Select(b => b.Definition).Distinct().Count() != m.Bindings.Length
57             || m.Bindings.Any(b => string.IsNullOrWhiteSpace(b.Definition) || b.Column <= 0 || !m.Frames.Any(f => f.Column == b.Column)
58                 || (b.OpenColumn is { } open && (open <= 0 || !m.Frames.Any(f => f.Column == open)))
59                 || b.BaseCutouts is { } cuts && cuts.Any(r => !Rect(r))

64     internal PaperdollLayer[] Layers(SpatialSimulation sim)
65     {
66         var root = sim.Inventory.Content!.Roots.Single(r => r.ActorId == sim.Player.Id);
67         if (root.ActorId != _manifest.Actor || root.BodyFamily != _manifest.BodyFamily) return [];
68         var result = new List<PaperdollLayer> { new(null, 0) };
69         foreach (var item in sim.BodyFitItems.Where(i => i.Placement.ParentId == root.Id
70             && sim.Inventory.Region(i.Placement.ParentId, i.Placement.RegionId).MountSlot is { } slot
71             && InventoryWear.Accepts(sim.Inventory.Definition(i.Id), slot)).OrderBy(i => i.Placement.WornOrder))
72         {
73             var binding = _manifest.Bindings.SingleOrDefault(b => b.Definition == sim.Inventory.Definition(item.Id).Id);
74             var shape = InventoryBodyFit.Shape(sim.Inventory, item.Id);
75             if (binding is null || shape is null || InventoryBodyFit.Pose(sim.Inventory, item) != shape.DefaultPose) continue;
76             var column = item.GarmentOpen ? binding.OpenColumn ?? binding.Column : binding.Column;
77             if (binding.MountColumns is { } mounts)
78             {
79                 var slot = sim.Inventory.Region(item.Placement.ParentId, item.Placement.RegionId).MountSlot!;
80                 if (!mounts.TryGetValue(slot, out column)) continue; // Unqualified mounts stay explicitly omitted.
81             }
82             result.Add(new(item.Id, column));
83             result.AddRange(AttachedLayers(sim, item, binding));
84         }

101     /// <summary>Only worn paperdoll parts use these cached derived pixels. Loose art/drag previews never consult this cache.</summary>
102     private PaperdollRaster Raster(PaperdollLayer layer, PaperdollPart part, string side)
103     {
104         var file = Frame(layer).File; var source = Source(part, side);
105         var cuts = side == "back" ? part.BackWornCutouts : part.FrontWornCutouts;
106         if (cuts is not { Length: > 0 }) return new(_textures[file], _images[file], source);
107         if (_worn.TryGetValue((part, side), out var cached)) return cached;
108         var pixels = _images[file].GetRegion(new((int)source.Position.X, (int)source.Position.Y, (int)source.Size.X, (int)source.Size.Y));
109         PaperdollWornMask.Apply(pixels, cuts);
110         var raster = new PaperdollRaster(ImageTexture.CreateFromImage(pixels), pixels, new(Vector2.Zero, source.Size));
111         _worn.Add((part, side), raster); return raster;
112     }

118     internal bool Opaque(PaperdollLayer layer, string side, Vector2 uv, bool worn = true)
119     {
120         foreach (var part in Parts(layer, side))
121         {
122             var dest = Destination(part, side); var local = layer.Mounted is { } mounted
123                 ? MountedLocal(mounted, dest, uv) : (uv - dest.Position) / dest.Size;
124             if (local.X is < 0 or >= 1 || local.Y is < 0 or >= 1) continue;
125             var raster = worn ? Raster(layer, part, side) : new PaperdollRaster(_textures[Frame(layer).File], _images[Frame(layer).File], Source(part, side));
126             var pixel = raster.Source.Position + local * raster.Source.Size;
127             if (raster.Pixels.GetPixel((int)pixel.X, (int)pixel.Y).A > .1f) return true;
128         }
129         return false;
130     }

139     /// <summary>Authored stand-in body cutouts prevent bare arms clipping through opaque long sleeves. No physical surface is changed.</summary>
140     private Rect2[] BaseRegions(PaperdollLayer[] layers)
141     {
142         var columns = layers.Where(l => l.ItemId is not null).Select(l => l.Column).ToHashSet();
143         var cuts = _manifest.Bindings.Where(b => columns.Contains(b.Column) || b.OpenColumn is { } open && columns.Contains(open))
144             .SelectMany(b => b.BaseCutouts ?? []).Select(r => new Rect2(r[0], r[1], r[2], r[3])).Distinct().ToArray();
145         if (cuts.Length == 0) return [new(0, 0, 1, 1)];
146         var xs = cuts.SelectMany(r => new[] { r.Position.X, r.End.X }).Concat(new float[] { 0, 1 }).Distinct().Order().ToArray();
147         var ys = cuts.SelectMany(r => new[] { r.Position.Y, r.End.Y }).Concat(new float[] { 0, 1 }).Distinct().Order().ToArray();
148         var result = new List<Rect2>();
149         for (var x = 0; x < xs.Length - 1; x++)
150         for (var y = 0; y < ys.Length - 1; y++)
151         {
152             var r = new Rect2(xs[x], ys[y], xs[x + 1] - xs[x], ys[y + 1] - ys[y]);
153             if (!cuts.Any(c => c.HasPoint(r.GetCenter()))) result.Add(r);
154         }
155         return result.ToArray();
156     }

161     private void DrawMosaic(Control canvas, Rect2 bounds, string side)
162     {
163         var body = Frame(new(null, 0)); var part = body.Parts.Single();
164         var source = Source(part, side); var dest = Destination(part, side);
165         foreach (var tile in PaperdollMosaic.Tiles(_manifest.Mosaic, side, new(_manifest.CanvasSize[0], _manifest.CanvasSize[1])))
166         {
167             var sample = source.Position + (tile.GetCenter() - dest.Position) / dest.Size * source.Size;
168             var color = _images[body.File].GetPixel((int)sample.X, (int)sample.Y);
169             // Opaque coarse blocks are synthesized by Godot; no mosaic is stored in the source image.
170             color = new(MathF.Round(color.R * 12) / 12, MathF.Round(color.G * 12) / 12, MathF.Round(color.B * 12) / 12, 1);
171             canvas.DrawRect(new(bounds.Position + tile.Position * bounds.Size, tile.Size * bounds.Size), color);
172         }
173     }
174     internal void Draw(Control canvas, SpatialSimulation sim, string side, string? hover, string? selected)
175     {
176         var layers = Layers(sim); if (layers.Length == 0) return;
177         var bounds = Bounds(canvas.Size);
178         foreach (var layer in layers)
179         {
180             var tint = layer.ItemId is not null && layer.ItemId == hover ? new Color(1.3f, 1.5f, 1.25f)
181                 : layer.ItemId is not null && layer.ItemId == selected ? new Color(1.15f, 1.25f, 1.1f) : Colors.White;
182             foreach (var part in Parts(layer, side))
183             {
184                 var dest = Destination(part, side); var raster = Raster(layer, part, side); var sourcePart = raster.Source;
185                 if (layer.Mounted is { } mounted)
186                 { DrawMounted(canvas, bounds, dest, raster.Texture, raster.Source, mounted, tint); continue; }
187                 foreach (var clip in layer.ItemId is null ? BaseRegions(layers) : new[] { dest })
188                 {
189                     var visible = dest.Intersection(clip); if (!visible.HasArea()) continue;
190                     var sourceClip = new Rect2(sourcePart.Position + (visible.Position - dest.Position) / dest.Size * sourcePart.Size,
191                         visible.Size / dest.Size * sourcePart.Size);
192                     canvas.DrawTextureRectRegion(raster.Texture, new(bounds.Position + visible.Position * bounds.Size, visible.Size * bounds.Size), sourceClip, tint);
193                 }
194             }
195             if (layer.ItemId is null) DrawMosaic(canvas, bounds, side);
196         }
197         var body = new PaperdollLayer(null, 0); var bodyPart = Frame(body).Parts.Single();
198         var source = Source(bodyPart, side); var destBody = Destination(bodyPart, side); var head = Head(side);
199         var texture = _textures[Frame(body).File];
200         canvas.DrawPolygon(head.Select(uv => bounds.Position + uv * bounds.Size).ToArray(), new[] { Colors.White },
201             head.Select(uv => (source.Position + (uv - destBody.Position) / destBody.Size * source.Size) / texture.GetSize()).ToArray(), texture);
202     }
