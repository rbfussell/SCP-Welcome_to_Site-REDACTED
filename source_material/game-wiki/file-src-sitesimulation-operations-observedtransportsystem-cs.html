37     public static void Advance(SimulationWorld world)
38     {
39         var state = world.Operations;
40         if (state.Definition is not { } d) return;

42             .OrderBy(o => o.Id).FirstOrDefault();
43         if (order is null)
44         {
45             ClearWork(world); state.Trolley = state.Trolley with { HandlerId = null }; return;
46         }
47         var actor = world.Actors.Single(a => a.Id == order.ActorId);
48         var body = Body(world);
49         var cart = state.Trolley;
50         if (cart.HandlerId is not null && cart.HandlerId != actor.Id) { ClearWork(world); cart = state.Trolley; }
51         state.Trolley = cart = cart with { HandlerId = actor.Id };
52         var cartCell = Hex.FromWorld(cart.Position);
53         if (cart.SubjectId is not null && body.Cell != cartCell) { DetachRelocated(world, body.Id); return; }
54         if (actor.Moving) return;
55         if (actor.StoppedStep is not null || actor.Cell.Distance(cartCell) != 1)
56         {
57             ClearWork(world);
58             var approach = Enumerable.Range(0, 6).Select(cartCell.Neighbor)
59                 .Where(c => world.Map.Walkable(c) && !MovementSystem.Occupied(world, actor).Contains(c))
60                 .Select(c => (Cell: c, Path: MovementSystem.PlanRoute(world, actor, c)))
61                 .Where(p => p.Path.Count > 0).OrderBy(p => p.Path.Count).FirstOrDefault();
62             if (approach.Path is null) { PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "Cannot reach trolley handle"); return; }
63             PersonnelOrderSystem.Travel(world, order, actor, approach.Cell, false); return;
64         }
65         var knownPoint = state.Contacts.SingleOrDefault(c => c.ActorId == actor.Id)?.Position ?? order.TargetPoint;
66         actor.Facing = ObservationService.FacingToward(actor.Position, cart.SubjectId is null ? knownPoint ?? cart.Position : cart.Position, actor.Facing);
67         if (world.LocalMs < actor.StartledUntilMs || !world.Anomalies.AllowsObservedTransport(body.Id))
68         { ClearWork(world); PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "Handler not ready or module disallows transport"); return; }

70         var seesBody = ObservationService.Inspect(world.Map, actor, body.Position, world.LocalMs).Detected;
71         if (cart.SubjectId is not null && !ObservedByOther(world, actor.Id))
72         { ClearWork(world); PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "Observation coverage lost · trolley stopped"); return; }
73         if (cart.SubjectId is null && seesBody && body.Cell.Distance(cartCell) == 1)
74         {
75             if (!ObservedByOther(world, actor.Id) || !world.Map.CanTraverse(body.Cell, cartCell, new HashSet<Hex> { body.Cell, cartCell }))
76             { ClearWork(world); PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "Loading needs clear access and another observer"); return; }
77             if (!Work(world, order, actor, cartCell, actor.Cell, "load")) return;
78             world.Map.DynamicBlockers.Remove(body.Cell);
79             body.Cell = cartCell; // External transport preserves anomalous appearance/pose/encounter identity.
80             world.Map.DynamicBlockers.Add(cartCell);
81             state.Trolley = state.Trolley with { SubjectId = body.Id, Step = null };
82             world.Log("transport", body.Id, "loaded", "Observed external loading onto open trolley; module remains active.");
83             return;
84         }
85         Hex destination;
86         if (cart.SubjectId is not null)
87         {
88             destination = d.Home;
89             if (cartCell == destination)
90             {
91                 // Unload into the receiving footprint; the empty trolley is parked at the handler's former hex.
92                 if (!ObservedByOther(world, actor.Id)) return;
93                 var parkedAt = actor.Cell;
94                 var exit = Enumerable.Range(0, 6).Select(actor.Cell.Neighbor).FirstOrDefault(c => c != cartCell && world.Map.Walkable(c)
95                     && !MovementSystem.Occupied(world, actor).Contains(c) && world.Map.CanTraverse(actor.Cell, c, new HashSet<Hex> { actor.Cell }));
96                 if (exit == default) { PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "No space to park trolley and step clear"); return; }
97                 if (!Work(world, order, actor, cartCell, actor.Cell, "unload")) return;
98                 world.Map.DynamicBlockers.Add(parkedAt);
99                 state.Trolley = new(parkedAt.World);
100                 actor.Cell = exit;
101                 actor.Facing = ObservationService.FacingToward(actor.Position, body.Position, actor.Facing);
102                 PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Complete, "Delivered and unloaded in receiving footprint");
103                 if (EmploymentSystem.Owns(world, actor.Id))
104                 {
105                     world.Log("employment", actor.Id, "order-complete", "Transport complete; return to scheduled duty on next tick.");
106                     return;
107                 }
108                 if (state.Orders.Count >= 32) state.Orders.RemoveAt(state.Orders.FindIndex(o => o.Status is OrderStatus.Complete or OrderStatus.Cancelled));
109                 state.Orders.Add(new(state.NextId++, order.Issuer, actor.Id, PersonnelTask.Observe, exit, order.Channel, OrderStatus.Active,
110                     "Procedure follow-up: maintain observation until relieved", world.LocalMs, body.Position));
111                 world.Log("transport", body.Id, "unloaded", "One body retained; empty trolley parked alongside receiving footprint.");
112                 return;
113             }
114         }
115         else
116         {
117             var contact = state.Contacts.SingleOrDefault(c => c.ActorId == actor.Id);
118             var point = seesBody ? body.Position : contact?.Position ?? order.TargetPoint;
119             if (point is null) { ClearWork(world); PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "No acquired location for subject"); return; }
120             if (!seesBody && cartCell.Distance(Hex.FromWorld(point.Value)) <= 1)
121             { ClearWork(world); PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "Last known area reached · reacquire subject"); return; }
122             var options = world.Map.Cells.Where(c => c.Distance(Hex.FromWorld(point.Value)) == 1 && world.Map.Walkable(c)
123                 && c != actor.Cell && !MovementSystem.Occupied(world, actor).Contains(c))
124                 .Select(c => (Cell: c, Path: CartPath(world, cartCell, c, actor))).Where(p => p.Path.Count > 0).OrderBy(p => p.Path.Count).FirstOrDefault();
125             if (options.Path is null) { ClearWork(world); PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "Last known area reached · reacquire subject or clear route"); return; }

127         }
128         var path = CartPath(world, cartCell, destination, actor);
129         if (path.Count == 0) { ClearWork(world); PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "Trolley route blocked · open access and clear personnel"); return; }

132         if (!world.Map.CanTraverse(cartCell, next, ignored) || !world.Map.CanTraverse(actor.Cell, cartCell, ignored)
133             || next == actor.Cell || MovementSystem.Occupied(world, actor).Contains(next))
134         { ClearWork(world); PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Blocked, "Push obstructed · reposition handler or clear next hex"); return; }
135         if (!Work(world, order, actor, next, cartCell, "push")) return;
136         world.Map.DynamicBlockers.Remove(cartCell);
137         world.Map.DynamicBlockers.Add(next);
138         actor.Cell = cartCell; actor.WalkMs += WorkMs;
139         if (state.Trolley.SubjectId is not null) body.Cell = next;
140         actor.Facing = ObservationService.FacingToward(actor.Position, next.World, actor.Facing);
141         state.Trolley = state.Trolley with { Position = next.World, Step = null };
142         world.Log("transport", actor.Id, "pushed", $"Trolley {cartCell} -> {next}; physical path validated; 600ms work.");

144     private static IReadOnlyList<Hex> CartPath(SimulationWorld world, Hex start, Hex end, ActorState handler)
145     {
146         var ignored = OwnCells(world.Operations);
147         if (world.Operations.Trolley.Step is { Stage: "push" } step) ignored.Add(step.Next);
148         return world.Map.FindPath(start, end, MovementSystem.Occupied(world, handler).Append(handler.Cell).ToHashSet(), ignored);
149     }
150     private static bool Work(SimulationWorld world, PersonnelOrder order, ActorState actor, Hex next, Hex handlerNext, string stage)
151     {
152         var state = world.Operations;
153         var step = state.Trolley.Step;
154         if (step is null || step.Stage != stage || step.Next != next)
155         {
156             ClearWork(world);
157             step = new(next, handlerNext, 0, stage);
158             if (stage == "push") world.Map.DynamicBlockers.Add(next);
159         }
160         step = step with { ProgressMs = step.ProgressMs + SpatialSimulation.TickMs };
161         state.Trolley = state.Trolley with { Step = step };
162         PersonnelOrderSystem.Set(world, order.Id, OrderStatus.Active, stage == "load" ? "Loading under observation" : stage == "unload" ? "Unloading in receiving footprint" : "Pushing trolley");
163         if (step.ProgressMs < WorkMs) return false;
164         ClearWork(world);
165         return true;
166     }
