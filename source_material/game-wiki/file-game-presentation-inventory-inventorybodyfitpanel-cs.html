151     private Vector2 Center(BodyHex cell) => new(Size.X / 2 + Mirror * (float)(Math.Sqrt(3) * Radius * (cell.Q + cell.R / 2.0)), 16 + Radius * 1.5f * cell.R);

154         + Vector2.FromAngle(Mathf.Pi / 6 + i * Mathf.Pi / 3) * (Radius - .45f)).ToArray();
155     private (Vector2 A, Vector2 B) Edge(Vector2 center, int direction)
156     {
157         var normal = (direction - 1) * Mathf.Pi / 3;
158         var a = Vector2.FromAngle(normal - Mathf.Pi / 6) * Radius; var b = Vector2.FromAngle(normal + Mathf.Pi / 6) * Radius;
159         return (center + new Vector2(Mirror * a.X, a.Y), center + new Vector2(Mirror * b.X, b.Y));
160     }
