20     private static AudioStreamWav Tone(double seconds, bool loop)
21     {
22         const int rate = 22050;
23         var count = (int)(seconds * rate);
24         var data = new byte[count * 2];
25         uint noise = 650;
26         double filtered = 0;
27         for (var i = 0; i < count; i++)
28         {
29             var t = i / (double)rate;
30             noise ^= noise << 13; noise ^= noise >> 17; noise ^= noise << 5;
31             filtered = filtered * 0.92 + ((noise / (double)uint.MaxValue) * 2 - 1) * 0.08;
32             var sample = loop
33                 ? 0.12 * Math.Sin(Math.Tau * 55 * t) + 0.04 * Math.Sin(Math.Tau * 110 * t) + filtered * 0.025
34                 : (0.22 * Math.Sin(Math.Tau * (52 * t - 12 * t * t)) + filtered * 0.35)
35                     * Math.Min(1, t / 0.018) * Math.Exp(-t * 7) * Math.Min(1, (seconds - t) / 0.05);
36             var pcm = (short)(Math.Clamp(sample, -0.35, 0.35) * short.MaxValue);
37             data[i * 2] = (byte)(pcm & 255); data[i * 2 + 1] = (byte)((pcm >> 8) & 255);
38         }
39         return new AudioStreamWav
40         {
41             Data = data, Format = AudioStreamWav.FormatEnum.Format16Bits, MixRate = rate,
42             LoopMode = loop ? AudioStreamWav.LoopModeEnum.Forward : AudioStreamWav.LoopModeEnum.Disabled,
43             LoopBegin = 0, LoopEnd = count
44         };
45     }
