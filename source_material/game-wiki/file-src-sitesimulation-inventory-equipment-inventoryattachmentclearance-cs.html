3 /// <summary>Checks the whole exterior assembly, including cousins on separate hooks or worn providers.</summary>
4 internal static class InventoryAttachmentClearance
5 {
6     private sealed record Part(InventoryItem Item, AttachmentFrame Frame, HashSet<BodyHex> Occupancy, HashSet<BodyHex> Contacts);
7 
8     internal static void Validate(InventoryState state)
9     {
10         var parts = state.Items.Values.Where(i => i.Placement.AttachmentPose is not null).Select(item =>
11         {
12             var p = item.Placement;
13             var support = state.Region(p.ParentId, p.RegionId).Attachment!;
14             var provider = InventoryAttachmentFrames.Provider(state, p.ParentId, support.Surface);
15             var frame = provider with { Pose = InventoryAttachmentFrames.Compose(provider.Pose, p.AttachmentPose!) };
16             var profile = state.Definition(item.Id).Attachment!;
17             return new Part(item, frame,
18                 profile.Occupancy.Select(c => BodyFitGeometry.Transform(c, frame.Pose)).ToHashSet(),
19                 profile.Contacts.Select(c => BodyFitGeometry.Transform(c, frame.Pose)).ToHashSet());
20         }).OrderBy(p => p.Item.Id, StringComparer.Ordinal).ToArray();
21         for (var a = 0; a < parts.Length; a++)
22         for (var b = a + 1; b < parts.Length; b++)
23         {
24             var left = parts[a]; var right = parts[b];
25             if (left.Frame.AnchorId != right.Frame.AnchorId || left.Frame.Surface != right.Frame.Surface) continue;
26             var overlap = left.Occupancy.Intersect(right.Occupancy).ToArray();
27             if (overlap.Length == 0) continue;
28             // A connector may meet its immediate provider at its own required contacts, never along its whole silhouette.
29             if (left.Item.Placement.ParentId == right.Item.Id && overlap.All(left.Contacts.Contains)
30                 || right.Item.Placement.ParentId == left.Item.Id && overlap.All(right.Contacts.Contains)) continue;
31             throw new ArgumentException("Attachment clearance: " + state.Definition(left.Item.Id).Name + " overlaps "
32                 + state.Definition(right.Item.Id).Name + " on the same exterior surface.");
33         }
34     }
35 }
