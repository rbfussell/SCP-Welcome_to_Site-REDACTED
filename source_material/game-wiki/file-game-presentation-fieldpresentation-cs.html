14     /// <summary>Cosmetic fade inside the real sight cone. Never used to decide whether an entity is seen.</summary>
15     public static float SightShade(SpatialSimulation sim, Point2 point)
16     {
17         if (!sim.VisibleToPlayer(point)) return 0;
18         var delta = point - sim.Player.Position;
19         if (delta.Length < 0.05) return 1;
20         var direction = sim.Player.Cell.Neighbor(sim.Player.Facing).World - sim.Player.Cell.World;
21         var dot = (delta.X * direction.X + delta.Y * direction.Y) / (delta.Length * direction.Length);
22         static double Fade(double value)
23         { var t = Math.Clamp(value, 0, 1); return t * t * (3 - 2 * t); }
24         return (float)(Fade((dot - ObservationService.FacingDotThreshold) / 0.22)
25             * Fade((ObservationService.VisionRange - delta.Length) / 0.7));
26     }
