6     public static InventoryPackingPlan Quote(SimulationWorld world, string actorId, InventoryRequest request, out int duration, out int ap)
7     {
8         duration = ap = 0;
9         if (request.Destination is not { X: 0, Y: 0, Rotated: false, WornOrder: null, BodyPose: null, AttachmentPose: null, GroundOrder: null } target || request.RegionId is not null)
10             throw new ArgumentException("Choose a compartment for packing.");
11         var state = world.Inventory;
12         if (request.Kind == InventoryActionKind.Arrange && request.ItemId != target.ParentId)
13             throw new ArgumentException("Arrange addresses one compartment of its owner.");
14         var access = InventoryAccessService.Placement(world, actorId, target);
15         if (!access.Allowed) throw new ArgumentException(access.Reason);
16         var incoming = request.Kind == InventoryActionKind.AutoPlace ? request.ItemId : null;
17         if (incoming is not null)
18         {
19             if (!state.Items.TryGetValue(incoming, out var item)) throw new ArgumentException("Unknown incoming item.");
20             var source = InventoryAccessService.Placement(world, actorId, item.Placement);
21             if (!source.Allowed) throw new ArgumentException(source.Reason);
22             if (InventoryGround.Covering(state, item).FirstOrDefault() is { } cover)
23                 throw new ArgumentException("Covered by " + state.Name(cover.Id) + "; move covering items first");
24             InventoryWear.ValidateAction(state, item, item.Placement);
25             if (item.Placement.ParentId == target.ParentId && item.Placement.RegionId == target.RegionId)
26                 throw new ArgumentException("Item is already in this compartment; place it on the grid or choose Arrange explicitly.");
27             if (item.Id == target.ParentId) throw new ArgumentException("A container cannot be placed inside itself.");
28             for (var parent = target.ParentId; state.Items.TryGetValue(parent, out var ancestor); parent = ancestor.Placement.ParentId)
29                 if (ancestor.Id == item.Id) throw new ArgumentException("A container cannot be placed inside its descendant.");
30         }
31         var plan = InventoryPacking.Plan(state, target.ParentId, target.RegionId, incoming);
32         foreach (var move in plan.Moves)
33         {
34             var definition = state.Definition(move.ItemId);
35             duration += definition.HandleMs + InventoryAccessService.HandlingMs(state, move.From) + InventoryAccessService.HandlingMs(state, move.To);
36             ap += definition.HandleAp + InventoryActions.AccessAp(state, move.From) + InventoryActions.AccessAp(state, move.To);
37         }
38         if (duration > 100000) throw new ArgumentException("Too much packing for one supported handling action; use smaller compartments.");
39         return plan;
40     }
