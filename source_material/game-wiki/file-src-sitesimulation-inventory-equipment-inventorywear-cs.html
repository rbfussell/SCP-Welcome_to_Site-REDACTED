31     private static InventoryItem? Blocker(InventoryState state, InventoryItem item, ItemPlacement placement, IEnumerable<string> regions,
32         Obstruction purpose)
33     {
34         var order = placement.WornOrder ?? InventoryWornOrder.Order(item);
35         return Worn(state, placement.ParentId).Where(other => other.Id != item.Id && InventoryWornOrder.Order(other) > order)
36             .Where(other => purpose switch
37             {
38                 Obstruction.Access => state.Definition(other.Id).Wear!.CoversAccess,
39                 Obstruction.Removal => state.Definition(other.Id).Wear!.BlocksRemoval,
40                 _ => true
41             })
42             .OrderByDescending(InventoryWornOrder.Order).ThenBy(other => other.Id, StringComparer.Ordinal)
43             .FirstOrDefault(other => Interferes(state, other, item, regions, purpose));

45     private static bool Interferes(InventoryState state, InventoryItem outer, InventoryItem inner, IEnumerable<string> regions, Obstruction purpose)
46     {
47         var wear = state.Definition(inner.Id).Wear!;
48         if (wear.OverlayLimit is not null) return InventoryWornStack.OverlayIntersects(state, outer, inner, purpose != Obstruction.Removal);
49         if (wear.Outermost && !InventoryWornStack.OverlayIntersects(state, outer, inner, purpose != Obstruction.Removal)) return false;
50         var blockedRegions = purpose == Obstruction.Removal
51             ? state.Definition(outer.Id).Wear!.RemovalRegions ?? Mount(state, outer, outer.Placement)!.Coverage : Covered(state, outer);
52         return blockedRegions.Intersect(regions).Any();

55     /// <summary>Partial exposure is enough to draw a garment; manipulating it can still require removing outer layers.</summary>
56     public static bool Visible(InventoryState state, InventoryItem item)
57     {
58         if (Mount(state, item, item.Placement) is not { } mount) return true;
59         var cells = InventoryWornStack.Cells(state, item);
60         if (cells.Length > 0)
61         {
62             var occluded = Worn(state, item.Placement.ParentId).Where(i => InventoryWornOrder.Order(i) > InventoryWornOrder.Order(item))
63                 .SelectMany(i => InventoryWornStack.Cells(state, i, true)).ToHashSet();
64             return cells.Any(c => !occluded.Contains(c));
65         }
66         return mount.Coverage.Any(region => Blocker(state, item, item.Placement, [region], Obstruction.Visual) is null);

99     internal static void ValidateGraph(InventoryState state)
100     {
101         foreach (var root in state.Content!.Roots.Where(r => r.ActorId is not null))
102         {
103             var worn = Worn(state, root.Id).OrderBy(i => i.Id, StringComparer.Ordinal).ToArray();
104             for (var a = 0; a < worn.Length; a++)
105             for (var b = a + 1; b < worn.Length; b++)
106             {
107                 var left = worn[a]; var right = worn[b];
108                 if (!InventoryWornStack.Overlaps(state, left, right)) continue;
109                 var inner = InventoryWornOrder.Order(left) < InventoryWornOrder.Order(right) ? left : right;
110                 if (state.Definition(inner.Id).Wear!.Outermost)
111                     throw new ArgumentException("Nothing can cover outermost " + state.Definition(inner.Id).Name + "; remove it first");
112             }
113             _ = InventoryWornStack.Build(state, root.Id);
114             InventoryWearFit.ValidateWorn(state, worn);
115         }
