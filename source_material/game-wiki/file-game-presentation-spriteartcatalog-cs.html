12 /// <summary>Presentation-only registration: atlas bounds, feet anchors, rig/state/facing and ordered layers.</summary>
13 public sealed class SpriteArtCatalog

15     public SpriteArtManifest Manifest { get; }
16     public SpriteArtCatalog(string json)
17     {
18         Manifest = JsonSerializer.Deserialize<SpriteArtManifest>(json, new JsonSerializerOptions { PropertyNameCaseInsensitive = true })
19             ?? throw new ArgumentException("Empty sprite-art manifest.");
20         var m = Manifest;
21         static bool FileName(string path) => !string.IsNullOrWhiteSpace(path) && !path.StartsWith('/') && !path.Contains('\\')
22             && !path.Contains(':') && path.Split('/').All(part => part is not ("" or "." or ".."));
23         if (m.Version != 1 || m.Images is null || m.Sprites is null || m.Rigs is null || m.Actors is null || m.Objects is null || m.Surfaces is null
24             || m.Images.Any(e => !FileName(e.Key) || e.Value is not { Length: 2 } || e.Value.Any(n => n <= 0)))
25             throw new ArgumentException("Invalid sprite-art schema or image declaration.");
26         if (m.Mounts is not null && m.Mounts.Any(e => string.IsNullOrWhiteSpace(e.Key) || e.Value is not { Length: 2 } || e.Value.Any(n => !double.IsFinite(n))))
27             throw new ArgumentException("Invalid presentation mount offset.");
28         foreach (var (key, s) in m.Sprites)
29             if (string.IsNullOrWhiteSpace(key) || s is null || !m.Images.TryGetValue(s.File, out var size)
30                 || s.Region is not { Length: 4 } || s.Region[0] < 0 || s.Region[1] < 0 || s.Region[2] <= 0 || s.Region[3] <= 0
31                 || (long)s.Region[0] + s.Region[2] > size[0] || (long)s.Region[1] + s.Region[3] > size[1]
32                 || s.Anchor is not { Length: 2 } || s.Anchor.Any(n => !double.IsFinite(n))
33                 || s.Anchor[0] < 0 || s.Anchor[0] > s.Region[2] || s.Anchor[1] < 0 || s.Anchor[1] > s.Region[3]
34                 || !double.IsFinite(s.Scale) || s.Scale <= 0)
35                 throw new ArgumentException("Invalid sprite region, scale or ground anchor: " + key);
36         foreach (var (key, rig) in m.Rigs)
37             if (string.IsNullOrWhiteSpace(key) || rig is null || rig.Directions is not { Length: 6 }
38                 || !rig.Directions.SequenceEqual(new[] { "NE", "E", "SE", "SW", "W", "NW" }) || rig.Frames is null
39                 || !rig.Frames.ContainsKey("idle") || !rig.Frames.ContainsKey("walk") || rig.Frames.Any(e => string.IsNullOrWhiteSpace(e.Key) || e.Value < 1))
40                 throw new ArgumentException("Invalid directional character rig: " + key);
41         foreach (var (id, actor) in m.Actors)
42         {
43             if (string.IsNullOrWhiteSpace(id) || actor is null || !m.Rigs.TryGetValue(actor.Rig, out var rig) || actor.Layers is not { Length: > 0 }
44                 || actor.Layers.Any(layer => layer is null || string.IsNullOrWhiteSpace(layer.Id) || layer.States is null)
45                 || actor.Layers.Select(layer => layer.Id).Distinct().Count() != actor.Layers.Length)
46                 throw new ArgumentException("Invalid actor appearance: " + id);
47             foreach (var layer in actor.Layers)
48             foreach (var (state, count) in rig.Frames)
49                 if (!layer.States.TryGetValue(state, out var directions) || directions is not { Length: 6 }
50                     || directions.Any(frames => frames is null || frames.Length != count || frames.Any(key => !m.Sprites.ContainsKey(key))))
51                     throw new ArgumentException($"Missing art binding: {id}/{layer.Id}/{state}");
52         }
53         if (m.Objects.Concat(m.Surfaces).Any(e => string.IsNullOrWhiteSpace(e.Key) || !m.Sprites.ContainsKey(e.Value)))
54             throw new ArgumentException("Unknown surface/object artwork.");
55     }
56     public string[] Layers(string actorId, string state, int facing, double phase)
57     {
58         if (!Manifest.Actors.TryGetValue(actorId, out var actor)) return [];
59         if (facing is < 0 or > 5 || !double.IsFinite(phase) || phase is < 0 or >= 1 || !Manifest.Rigs[actor.Rig].Frames.TryGetValue(state, out var count))
60             throw new ArgumentException("Invalid visual state/facing/phase.");
61         var frame = Math.Min(count - 1, (int)(phase * count));
62         return actor.Layers.Select(layer => layer.States[state][facing][frame]).ToArray();
63     }
