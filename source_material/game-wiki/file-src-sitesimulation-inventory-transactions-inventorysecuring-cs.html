11     public static bool Prepare(SimulationWorld world, IEnumerable<InventoryRequest> requests, out string reason, bool automatic = false)
12     {
13         reason = "Finish the current handling or securing sequence first.";
14         var state = world.Inventory;
15         if (!state.Enabled || state.Pending is not null || state.SecureQueue.Count != 0) return false;
16         var entries = requests.Take(MaxSteps + 1).ToArray();
17         if (entries.Length > MaxSteps || entries.Any(r => r is null || !IsSecure(r) || r.Destination is not null
18             || !state.Items.ContainsKey(r.ItemId) || (r.Kind == InventoryActionKind.Close) != (r.RegionId is not null)))
19         { reason = "Invalid securing targets."; return false; }
20         var steps = new List<InventorySecureStep>();
21         foreach (var request in entries.Distinct())
22         {
23             // Validate even completed requests structurally; only the already-state rejection is harmless.
24             var valid = InventoryActions.Validate(world, world.Player.Id, request, out reason, out _, out var ap);
25             if (!valid)
26             {
27                 if (reason is "Compartment is already in that state" or "Garment is already in that state") continue;
28                 return false;
29             }
30             if (world.Combat.Active && ap > CombatState.TurnAp)
31             { reason = $"This closure needs {ap} AP, more than one supported turn. Close deeper containers separately."; return false; }
32             steps.Add(new(request, state.Items[request.ItemId].Placement, automatic));
33         }
34         // Contained closures first; open inner garments before outer garments can cover their fastenings.
35         state.SecureQueue.AddRange(steps.OrderByDescending(s => Depth(state, s.Request.ItemId))
36             .ThenBy(s => s.Request.Kind == InventoryActionKind.CloseGarment ? 1 : 0)
37             .ThenBy(s => InventoryWornOrder.Order(state.Items[s.Request.ItemId])));
38         reason = steps.Count == 0 ? "These containers are already secured." : $"Securing queued: {steps.Count} closures; normal time/AP apply. Cancel handling to stop.";
39         if (steps.Count != 0) PlayerFeedback.Add(world, MessageKind.Equipment, "Inventory: " + reason);
40         return true;
41     }
42     private static int Depth(InventoryState state, string id)

48     public static void TryStart(SimulationWorld world)
49     {
50         var state = world.Inventory;
51         if (state.Pending is not null || state.SecureQueue.Count == 0 || world.LocalMs < world.Player.StartledUntilMs) return;
52         if (world.Combat.Active && !TacticalCombatSystem.CanAct(world, world.Player.Id, out _)) return;
53         var step = state.SecureQueue[0];
54         if (step.Automatic && !InventoryClosures.SecureOnClose(state, step.Request.ItemId, step.Request.RegionId))
55         { state.SecureQueue.RemoveAt(0); return; }
56         if (state.Items[step.Request.ItemId].Placement != step.Placement)
57         { Stop(world, "An inspected object moved; inspect it again before securing."); return; }
58         if (Completed(state, step.Request)) { state.SecureQueue.RemoveAt(0); return; }
59         if (!InventoryActions.Validate(world, world.Player.Id, step.Request, out var reason, out _, out var ap))
60         { Stop(world, reason); return; }
61         // Wait for a later player turn without ending the current turn or spending partial AP.
62         if (world.Combat.Active && world.Combat.For(world.Player.Id)!.Ap < ap) return;
63         if (!InventoryActions.Start(world, step.Request, out reason)) { Stop(world, reason); return; }
64         state.SecureQueue.RemoveAt(0);
65     }
