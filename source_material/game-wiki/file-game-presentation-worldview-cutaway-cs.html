20     {
21         var feet = Screen(Simulation.Player.Position);
22         var actor = Simulation.Player;
23         if (!_cutawayHeights.TryGetValue((actor.Id, DiagnosticSprites), out var height))
24         {
25             // Keep a stable body envelope over every facing/step, so the feather does not pulse while walking.
26             var keys = !DiagnosticSprites && _facilityArt.Manifest.Actors.TryGetValue(actor.Id, out var appearance)
27                 ? appearance.Layers.SelectMany(layer => layer.States.Values).SelectMany(directions => directions).SelectMany(frames => frames)
28                 : _sprites.Keys.Where(key => key.StartsWith(actor.Kind + "-", StringComparison.Ordinal));

31         }
32         _playerCutaway = PlayerWallCutaway.AroundSprite(new(feet.X, feet.Y), height);

48     /// <summary>Draw a textured quad with continuous vertex alpha; reuse its mesh until the mask or lighting changes.</summary>
49     private void FeatheredQuad(string id, Vector2[] corners, Color light, Texture2D? texture = null)
50     {
51         if (!_cutawayMeshes.TryGetValue(id, out var cached) || cached.Center != CutawayCenter
52             || cached.Radius != _playerCutaway.Radius || cached.Light != light || !cached.Corners.SequenceEqual(corners))
53         {
54             var mesh = cached?.Mesh ?? new ArrayMesh();
55             mesh.ClearSurfaces();
56             // Order: bottom-left, bottom-right, top-right, top-left, matching the normal wall UVs.
57             var columns = Math.Max(1, (int)Math.Ceiling(corners[0].DistanceTo(corners[1]) / CutawayVertexSpacing));
58             var rows = Math.Max(1, (int)Math.Ceiling(corners[0].DistanceTo(corners[3]) / CutawayVertexSpacing));
59             var vertices = new Vector2[(columns + 1) * (rows + 1)];
60             var colors = new Color[vertices.Length];
61             var uvs = new Vector2[vertices.Length];
62             for (var y = 0; y <= rows; y++)
63             for (var x = 0; x <= columns; x++)
64             {
65                 var u = x / (float)columns; var v = y / (float)rows;
66                 var point = corners[0].Lerp(corners[1], u).Lerp(corners[3].Lerp(corners[2], u), v);
67                 var index = y * (columns + 1) + x;
68                 vertices[index] = point;
69                 colors[index] = new Color(light, light.A * CutawayAlpha(point));
70                 uvs[index] = new(u, 1 - v);
71             }
72             var indices = new int[columns * rows * 6]; var next = 0;
73             for (var y = 0; y < rows; y++)
74             for (var x = 0; x < columns; x++)
75             {
76                 var a = y * (columns + 1) + x; var b = a + 1; var d = a + columns + 1; var c = d + 1;
77                 indices[next++] = a; indices[next++] = b; indices[next++] = c;
78                 indices[next++] = a; indices[next++] = c; indices[next++] = d;
79             }
80             using var arrays = new Godot.Collections.Array(); arrays.Resize((int)Mesh.ArrayType.Max);
81             arrays[(int)Mesh.ArrayType.Vertex] = vertices;
82             arrays[(int)Mesh.ArrayType.Color] = colors;
83             arrays[(int)Mesh.ArrayType.TexUV] = uvs;
84             arrays[(int)Mesh.ArrayType.Index] = indices;
85             mesh.AddSurfaceFromArrays(Mesh.PrimitiveType.Triangles, arrays, flags: Mesh.ArrayFormat.FlagUse2DVertices);
86             _cutawayMeshes[id] = new(mesh, CutawayCenter, _playerCutaway.Radius, light, corners);
87         }
88         DrawMesh(_cutawayMeshes[id].Mesh, texture, Transform2D.Identity);
89     }

91     private void WallStroke(Vector2 start, Vector2 end, Color color, float width, bool feather)
92     {
93         if (!feather) { DrawLine(start, end, color, width); return; }
94         var segments = Math.Max(1, (int)Math.Ceiling(start.DistanceTo(end) / 2));
95         for (var i = 0; i < segments; i++)
96         {
97             var a = start.Lerp(end, i / (float)segments); var b = start.Lerp(end, (i + 1) / (float)segments);
98             DrawLine(a, b, new Color(color, color.A * CutawayAlpha((a + b) * 0.5f)), width);
