39     private MountTarget? NearestMount(Vector2 at)
40     {
41         var providers = new[] { (Id: ItemId, Pose: new BodyPose(0, 0)) }.Concat(Mounted.Select(p => (Id: p.Item.Id, Pose: p.Pose)));
42         return providers.SelectMany(provider => (Panel.Simulation.Inventory.Definition(provider.Id).Regions ?? [])
43             .Where(r => r.Attachment is not null && Panel.Simulation.InventoryAccess(provider.Id, r.Id).Allowed)
44             .SelectMany(region => region.Attachment!.Cells.Select(cell => new MountTarget(provider.Id, region, cell, provider.Pose,
45                 MountPoint(BodyFitGeometry.Transform(cell, provider.Pose)).DistanceTo(at)))))
