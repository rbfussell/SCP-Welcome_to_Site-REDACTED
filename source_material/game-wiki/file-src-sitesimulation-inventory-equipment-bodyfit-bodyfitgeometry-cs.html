12     public static BodyHex Transform(BodyHex local, BodyPose pose) => local.Rotate(pose.Rotation) + new BodyHex(pose.Q, pose.R);
13     public static BodyFitResult Evaluate(BodyMask body, BodyFitShape shape, BodyPose pose)
14     {
15         if (pose.Q is < -64 or > 64 || pose.R is < -64 or > 64 || pose.Rotation is < 0 or > 5 || !shape.Rotations.Contains(pose.Rotation))
16             return new(false, "Unsupported body position or orientation.", [], [], [], []);
17         if (body.Id != shape.BodyFamily) return new(false, "Equipment was authored for a different body family.", [], [], [], []);
18         var bodyCells = Cells(body.Rows).ToHashSet();
19         var localCells = Cells(shape.Rows).Select(c => new BodyHex(c.Q - shape.DefaultPose.Q, c.R - shape.DefaultPose.R)).ToArray();
20         var localContacts = shape.Contacts;
21         if (shape.LinearSpan is { } span)
22         {
23             // Find the first exterior cell on each end. Independent extents allow a chest diagonal
24             // without shifting its chosen anchor. Never bridge a gap, shrink the core or exceed the authored range.
25             var low = localCells.Min(c => c.Q); var high = localCells.Max(c => c.Q);
26             while (high - low + 1 <= span.MaxCells && bodyCells.Contains(Transform(new(low - 1, 0), pose))) low--;
27             while (high - low + 1 <= span.MaxCells && bodyCells.Contains(Transform(new(high + 1, 0), pose))) high++;
28             if (high - low + 1 >= span.MinCells && high - low + 1 <= span.MaxCells)
29             {
30                 localCells = Enumerable.Range(low, high - low + 1).Select(q => new BodyHex(q, 0)).ToArray();
31                 localContacts = [new(new(low, 0), 4), new(new(high, 0), 1)];
32             }
33         }
34         var cells = localCells.Select(c => Transform(c, pose)).ToArray();
35         var outside = cells.Where(c => !bodyCells.Contains(c)).ToArray();
36         var contacts = localContacts.Select(edge => new BodyContact(Transform(edge.Cell, pose), (edge.Direction + pose.Rotation) % 6)).ToArray();
37         var failed = contacts.Where(edge => !bodyCells.Contains(edge.Cell) || bodyCells.Contains(edge.Cell.Neighbor(edge.Direction))).ToArray();
38         var fits = outside.Length == 0 && failed.Length == 0;
39         return new(fits, fits ? shape.LinearSpan is null ? "Shape and boundary contacts fit." : $"Strap fitted to {cells.Length} cells within authored {shape.LinearSpan.MinCells}–{shape.LinearSpan.MaxCells}; both ends contact."
40             : outside.Length > 0 ? $"{outside.Length} worn cells fall outside the body."
41             : $"{failed.Length} closed edges fail body contact; body continues through them.", cells, outside, contacts, failed);
42     }
