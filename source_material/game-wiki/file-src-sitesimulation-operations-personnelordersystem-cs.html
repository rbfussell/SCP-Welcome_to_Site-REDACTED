8     public static bool Issue(SimulationWorld world, string issuerId, string actorId, PersonnelTask task, Hex destination,
9         string channel, out string reason, bool allowPausedIntent = false)
10     {
11         var state = world.Operations;
12         var actor = world.Actors.SingleOrDefault(a => a.Id == actorId);
13         reason = !world.LocalActive && !(allowPausedIntent && world.ManualPause && !world.LocalHold && !world.Combat.Active) ? "Local simulation is held."
14             : state.Definition is null ? "No operation configured."
15             : state.Definition.CommanderId != issuerId ? "Issuer lacks site operations authority."
16             : actor is null || actor.Id == issuerId || actor.ObserverKind != "human" ? "Recipient is not an assigned human employee."
17             : !Enum.IsDefined(task) || channel is not ("voice" or "terminal" or "phone") ? "Unsupported task/channel."
18             : task == PersonnelTask.ResumeDuty && !EmploymentSystem.Owns(world, actorId) ? "Recipient has no employment appointment."
19             : task == PersonnelTask.Transport && state.Orders.Any(o => o.ActorId != actorId && o.Task == PersonnelTask.Transport
20                 && o.Status is OrderStatus.Pending or OrderStatus.Active or OrderStatus.Blocked) ? "Trolley already assigned to another handler."
21             : !world.Map.Walkable(destination) ? "Destination is blocked."
22             : channel == "terminal" && !FacilityCommunications.TerminalReachable(world) ? "Approach the powered, connected OSCP terminal."
23             : channel == "phone" && (!OscpService.PhoneAvailable(world) || !state.NetworkOnline) ? "A powered, accessible handset and site link are required."
24             : "accepted";
25         if (reason != "accepted") return false;
26         // A bounded history cannot evict an active order.
27         if (state.Orders.Count >= 32)
28         {
29             var completed = state.Orders.FindIndex(o => o.Status is OrderStatus.Complete or OrderStatus.Cancelled);
30             if (completed < 0) { reason = "Order ledger is full."; return false; }
31             state.Orders.RemoveAt(completed);
32         }
33         state.Orders.Add(new(state.NextId++, issuerId, actorId, task, destination, channel, OrderStatus.Pending,
34             "Awaiting delivery and acknowledgement", world.LocalMs,
35             task == PersonnelTask.Follow ? world.Player.Position : world.Awareness.Evidence.SingleOrDefault(e => e.SubjectId == state.Definition!.SubjectId)?.Position));
36         world.Log("orders", actorId, "submitted", $"{task} via {channel}; destination {destination}");
37         return true;
38     }

51     public static void Advance(SimulationWorld world)

53         var state = world.Operations;
54         if (state.Definition is not { } d) return;
55         foreach (var pending in state.Orders.Where(o => o.Status == OrderStatus.Pending).ToArray())
56         {
57             if (!FacilityCommunications.Link(world, pending)) { PlayerFeedback.NoAcknowledgement(world, pending); continue; }
58             if (state.Orders.Any(o => o.ActorId == pending.ActorId && o.Task == PersonnelTask.Transport
59                 && o.Status is OrderStatus.Active or OrderStatus.Blocked)) ObservedTransportSystem.Interrupt(world);
60             foreach (var previous in state.Orders.Where(o => o.ActorId == pending.ActorId && o.Id < pending.Id
61                 && o.Status is OrderStatus.Active or OrderStatus.Blocked).ToArray())
62                 Set(world, previous.Id, OrderStatus.Cancelled, "Relieved by an acknowledged replacement order");
63             var actor = world.Actors.Single(a => a.Id == pending.ActorId);
64             MovementSystem.StopInPlace(actor);
65             actor.Attending = true;
66             Set(world, pending.Id, OrderStatus.Active, "Acknowledged"
67                 + (EmploymentSystem.Owns(world, actor.Id) && pending.Task != PersonnelTask.ResumeDuty ? " · reassignment may leave the employment post uncovered" : ""));
68             if (pending.Task == PersonnelTask.ResumeDuty)
69                 Set(world, pending.Id, OrderStatus.Complete, "Released from direct orders; resuming scheduled employment");
70         }
71         foreach (var order in state.Orders.Where(o => o.Status is OrderStatus.Active or OrderStatus.Blocked).ToArray())
72         {
73             var actor = world.Actors.Single(a => a.Id == order.ActorId);
74             if (world.LocalMs < actor.StartledUntilMs) { Set(world, order.Id, OrderStatus.Blocked, "Recovering from startle"); continue; }
75             if (order.Task == PersonnelTask.Transport) continue; // The physical transport service owns this continuation.
76             if (order.Task == PersonnelTask.Post)
77             {
78                 if (actor.Cell == order.Destination && !actor.Moving && actor.StoppedStep is null)
79                 {
80                     if (order.TargetPoint is { } point) actor.Facing = ObservationService.FacingToward(actor.Position, point, actor.Facing);
81                     Set(world, order.Id, OrderStatus.Active, "Holding assigned post"); continue;
82                 }
83                 Travel(world, order, actor, order.Destination, false); continue;
84             }
85             Point2 target;
86             if (order.Task == PersonnelTask.Follow)
87             {
88                 var leader = world.Actors.Single(a => a.Id == order.Issuer);
89                 target = order.TargetPoint ?? actor.Position;
90                 actor.Facing = ObservationService.FacingToward(actor.Position, target, actor.Facing);
91                 if (ObservationService.Inspect(world.Map, actor, leader.Position, world.LocalMs).Detected)
92                 {
93                     target = leader.Position;
94                     var index = state.Orders.FindIndex(o => o.Id == order.Id);

96                 }
97                 else if (Point2.Distance(actor.Position, target) <= 1)
98                 { MovementSystem.StopInPlace(actor); Set(world, order.Id, OrderStatus.Blocked, "Leader not in sight; last known position reached"); continue; }
99                 if (Point2.Distance(actor.Position, target) <= 1)

101             }
102             else
103             {
104                 var contact = state.Contacts.SingleOrDefault(c => c.ActorId == actor.Id);
105                 // A delivered order shares the issuer's acquired point, never a hidden live transform.
106                 var deliveredPoint = order.TargetPoint;
107                 if (contact is null && deliveredPoint is null)
108                 { Set(world, order.Id, OrderStatus.Blocked, "No acquired sighting to investigate"); continue; }
109                 target = contact?.Position ?? deliveredPoint!.Value;
110                 actor.Facing = ObservationService.FacingToward(actor.Position, target, actor.Facing);
111                 var subject = world.Anomalies.Bodies.Single(b => b.Id == d.SubjectId);
112                 if (ObservationService.Inspect(world.Map, actor, subject.Position, world.LocalMs).Detected)
113                 {
114                     MovementSystem.StopInPlace(actor);
115                     Set(world, order.Id, OrderStatus.Active, "Maintaining direct observation"); continue;
116                 }
117                 if (ObservationService.Inspect(world.Map, actor, target, world.LocalMs).Detected)
118                 {
119                     MovementSystem.StopInPlace(actor);
120                     Set(world, order.Id, OrderStatus.Blocked, "Last known position empty · awaiting updated sighting"); continue;
121                 }
122             }
123             actor.Facing = ObservationService.FacingToward(actor.Position, target, actor.Facing);
124             var vantage = world.Map.Cells.Where(c => world.Map.Walkable(c) && Point2.Distance(c.World, target) is > 0.5 and < 2
125                 && world.Map.ClearSight(c.World, target) && !MovementSystem.Occupied(world, actor).Contains(c))
126                 .Select(c => (Cell: c, Path: MovementSystem.PlanRoute(world, actor, c)))
127                 .Where(p => p.Path.Count > 0).OrderBy(p => p.Path.Count).ThenBy(p => p.Cell.Q).ThenBy(p => p.Cell.R).FirstOrDefault();
128             if (vantage.Path is null) { Set(world, order.Id, OrderStatus.Blocked, "Last known area reached or no clear observation route"); continue; }
129             Travel(world, order, actor, vantage.Cell, true);

132     internal static void Travel(SimulationWorld world, PersonnelOrder order, ActorState actor, Hex destination, bool keepFacing)
133     {
134         if (actor.Moving) { Set(world, order.Id, OrderStatus.Active, "Travelling"); return; }
135         var path = MovementSystem.PlanRoute(world, actor, destination);
136         if (path.Count == 0) { Set(world, order.Id, OrderStatus.Blocked, "Route blocked · waiting for access or clearance"); return; }
137         MovementSystem.CommitRoute(actor, path, keepFacing);
138         Set(world, order.Id, OrderStatus.Active, "Travelling");
139     }
