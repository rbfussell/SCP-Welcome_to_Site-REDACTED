6 /// <summary>Explicit training fixture. The simulation alone grants turns and accepts AP transactions.</summary>
7 public partial class CombatPanel : VBoxContainer

12     private Button _arm = null!;
13     private OptionButton _opponent = null!;
14     private OptionButton _target = null!;
15     private string[] _opponents = [], _targets = [];
16     public void Initialize(SpatialSimulation sim)
17     {
18         _sim = sim;
19         AddThemeConstantOverride("separation", 5);
20         _status = new Label { AutowrapMode = TextServer.AutowrapMode.WordSmart }; AddChild(_status);
21         _opponent = new OptionButton { FocusMode = FocusModeEnum.None }; AddChild(_opponent);
22         _opponents = sim.Actors.Where(a => a != sim.Player).Select(a => a.Id).ToArray();
23         foreach (var id in _opponents) _opponent.AddItem(sim.Actors.Single(a => a.Id == id).Name + " · training opponent");
24         _opponent.Select(Array.IndexOf(_opponents, "tech"));
25         _arm = Button(this, "Arm nonlethal training drill", () => sim.ArmTraining(_opponents[_opponent.Selected]));
26         _actions = new VBoxContainer(); AddChild(_actions);
27         _target = new OptionButton { FocusMode = FocusModeEnum.None }; _actions.AddChild(_target);
28         Button(_actions, "Ready sidearm  ·  2 AP  [V]", () => sim.UseTrainingEquipment(CombatActionKind.ReadyWeapon));
29         Button(_actions, "Fire  ·  4 AP", () => { if (_target.Selected >= 0 && _target.Selected < _targets.Length) sim.FireTraining(_targets[_target.Selected]); });
30         Button(_actions, "Reload  ·  3 AP  [Y]", () => sim.UseTrainingEquipment(CombatActionKind.Reload));
31         Button(_actions, "End turn  [Enter]", () => sim.EndTurn());
32         Button(_actions, "End training drill", () => sim.EndTraining());
33         Refresh();
34     }
35     private static Button Button(Node parent, string text, Action action)
36     {
37         var button = new Button { Text = text, FocusMode = FocusModeEnum.None };
38         button.AddThemeFontSizeOverride("font_size", 12); button.Pressed += action; parent.AddChild(button); return button;
39     }
40     public void Refresh()
41     {
42         var combat = _sim.Combat; var player = combat.For(_sim.Player.Id);
43         _arm.Visible = _opponent.Visible = !combat.Armed;
44         _actions.Visible = combat.Armed;
45         _status.Text = !combat.Armed ? "TACTICAL TRAINING  /  provisional equipment"
46             : !combat.Active ? "DRILL ARMED · find the opponent by sight."
47             : $"TRAINING · ROUND {combat.Round}\n{(combat.CurrentActorId == _sim.Player.Id ? "YOUR TURN" : "OTHER TURN")} · {(combat.Pending is { } action ? action.Kind.ToString().ToUpperInvariant() : "CHOOSING")}\nAP {player!.Ap}/10 · points {player.Points}/6\nSidearm: {(player.WeaponReady ? "ready" : "stowed")} · {player.Ammo}/6 + {player.ReserveAmmo} rounds\nMove 1/hex · turn 1 · door 2 AP";
48         var targets = combat.Participants.Where(c => c.Side == "opposition" && c.Points > 0)
49             .Select(c => _sim.Actors.Single(a => a.Id == c.ActorId)).Where(a => _sim.VisibleToPlayer(a.Position)).Select(a => a.Id).ToArray();
50         if (!_targets.SequenceEqual(targets))
51         {
52             _targets = targets; _target.Clear();
53             foreach (var id in targets) _target.AddItem(_sim.Actors.Single(a => a.Id == id).Name);
54         }
55         _target.Disabled = targets.Length == 0;
56         _target.TooltipText = "Only directly visible training opponents appear here.";
57     }
