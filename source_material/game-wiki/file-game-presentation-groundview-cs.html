22     public override void _Ready()
23     {
24         ShowBehindParent = true;
25         TextureFilter = TextureFilterEnum.Nearest;
26         var map = Simulation.Map.Definition;
27         var tiles = from y in Enumerable.Range(-2, map.Height + 4)
28                     from x in Enumerable.Range(-2, map.Width + 4)
29                     let indoor = x >= 0 && y >= 0 && x < map.Width && y < map.Height
30                     let key = $"{(indoor ? "floor" : "land")}-{Math.Abs((x * 17 + y * 31) % 6)}"
31                     select (x, y, indoor, key);
32         foreach (var group in tiles.GroupBy(t => t.key))
33         {
34             var vertices = new List<Vector2>(); var uvs = new List<Vector2>(); var indices = new List<int>();
35             var planar = new List<Vector2>();
36             var ground = new List<Point2>(); var tints = new List<Color>();
37             var size = Textures[group.Key].GetSize();
38             foreach (var tile in group)
39             {
40                 var first = vertices.Count;
41                 var tint = tile.indoor && tile.y is >= 3 and < 5 ? new Color(1.09f, 1.03f, 0.86f) : Colors.White;
42                 for (var v = 0; v <= Subdivisions; v++)
43                 for (var u = 0; u <= Subdivisions; u++)
44                 {
45                     var offset = new Point2(u / (double)Subdivisions, v / (double)Subdivisions);
46                     var point = new Point2(tile.x + offset.X, tile.y + offset.Y);
47                     var uv = WorldView.Screen(offset) + new Vector2(0, 12);
48                     vertices.Add(WorldView.Screen(point)); uvs.Add(uv / size);
49                     planar.Add(new((float)offset.X, (float)offset.Y));
50                     ground.Add(point); tints.Add(tint);
51                 }
52                 for (var v = 0; v < Subdivisions; v++)
53                 for (var u = 0; u < Subdivisions; u++)
54                 {
55                     var a = first + v * (Subdivisions + 1) + u;
56                     var b = a + 1; var c = a + Subdivisions + 1; var d = c + 1;
57                     indices.AddRange([a, b, c, b, d, c]);
58                 }
59             }
60             _batches.Add(new(new ArrayMesh(), group.Key, Textures[group.Key], vertices.ToArray(), uvs.ToArray(), planar.ToArray(),
61                 indices.ToArray(), ground.ToArray(), tints.ToArray()));
62         }
63     }

65     public void Refresh(bool fieldView, bool authored = true)
66     {
67         _authored = authored;
68         var player = Simulation.Player;
69         var stamp = new SightStamp(player.Position, player.Cell, player.Facing, player.Attending,
70             Simulation.LocalMs >= player.EyesClosedUntilMs, Simulation.Map.LightsOn, fieldView, authored,
71             string.Join(',', Simulation.Map.OpenDoors.Order(StringComparer.Ordinal)));
72         if (stamp == _stamp) return;
73         _stamp = stamp;
74         var samples = new Dictionary<Point2, Color>();
75         var dim = new Color(0.28f, 0.32f, 0.34f); var lit = new Color(0.92f, 1, 0.96f);
76         foreach (var batch in _batches)
77         {
78             var colors = new Color[batch.Ground.Length];
79             for (var i = 0; i < colors.Length; i++)
80             {
81                 var point = batch.Ground[i];
82                 if (!samples.TryGetValue(point, out var light))
83                 {
84                     light = fieldView ? dim.Lerp(lit, FieldPresentation.SightShade(Simulation, point))
85                         : stamp.LightsOn ? Colors.White : new Color(0.3f, 0.34f, 0.4f);
86                     samples.Add(point, light);
87                 }
88                 colors[i] = light * batch.Tints[i];
89             }
90             using var arrays = new Godot.Collections.Array(); arrays.Resize((int)Mesh.ArrayType.Max);
91             arrays[(int)Mesh.ArrayType.Vertex] = batch.Vertices;
92             arrays[(int)Mesh.ArrayType.TexUV] = authored && AuthoredTextures.ContainsKey(batch.Key) ? batch.PlanarUVs : batch.UVs;
93             arrays[(int)Mesh.ArrayType.Color] = colors;
94             arrays[(int)Mesh.ArrayType.Index] = batch.Indices;
95             batch.Mesh.ClearSurfaces(); batch.Mesh.AddSurfaceFromArrays(Mesh.PrimitiveType.Triangles, arrays);
96         }
97         ShadingBuilds++;
98         QueueRedraw();
99     }
