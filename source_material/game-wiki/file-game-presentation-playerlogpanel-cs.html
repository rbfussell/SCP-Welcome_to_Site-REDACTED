17     {
18         MouseFilter = MouseFilterEnum.Stop;
19         AddThemeStyleboxOverride("panel", new StyleBoxFlat { BgColor = new("101a19"), BorderColor = new("415748"),
20             BorderWidthTop = 1, ContentMarginLeft = 10, ContentMarginRight = 8, ContentMarginTop = 6, ContentMarginBottom = 6 });
21         var row = new HBoxContainer(); AddChild(row);
22         var column = new VBoxContainer { SizeFlagsHorizontal = SizeFlags.ExpandFill }; row.AddChild(column);
23         column.AddThemeConstantOverride("separation", 1);
24         _heading = new Label { Text = "EVENT LOG  /  100 entries", MouseFilter = MouseFilterEnum.Ignore };
25         _heading.AddThemeFontSizeOverride("font_size", 11); column.AddChild(_heading);
26         for (var i = 0; i < _lines.Length; i++)
27         {
28             _lines[i] = new Label { Text = " ", TextOverrunBehavior = TextServer.OverrunBehavior.TrimEllipsis,
29                 CustomMinimumSize = new(0, 18), MouseFilter = MouseFilterEnum.Stop };
30             _lines[i].AddThemeFontSizeOverride("font_size", 12); column.AddChild(_lines[i]);
31         }
32         _scroll = new VScrollBar { CustomMinimumSize = new(16, 0), Step = 1, Page = 5 }; row.AddChild(_scroll);
33         _scroll.ValueChanged += value =>
34         {
35             if (_updating || _source is null) return;
36             _anchor = value >= Math.Max(0, _source.Entries.Count - 5) ? null : _source.Entries[(int)value].Id;
37             DrawEntries();
38         };
39         GuiInput += input =>
40         {
41             if (input is InputEventMouseButton { Pressed: true } mouse && mouse.ButtonIndex is MouseButton.WheelUp or MouseButton.WheelDown)
42             { _scroll.Value += mouse.ButtonIndex == MouseButton.WheelUp ? -3 : 3; AcceptEvent(); }
43         };
44     }
45     public void Refresh(PlayerEventLog source)
46     {
47         if (_source == source && _revision == source.Revision) return;
48         if (_source != source) _anchor = null;
49         _source = source; _revision = source.Revision; DrawEntries();
50     }
51     private void DrawEntries()
52     {
53         if (_source is null) return;
54         var entries = _source.Window(_anchor);
55         var start = entries.Length == 0 ? 0 : _source.Entries.ToList().FindIndex(e => e.Id == entries[0].Id);
56         _updating = true; _scroll.MaxValue = Math.Max(5, _source.Entries.Count); _scroll.Value = Math.Max(0, start); _updating = false;
57         _heading.Text = $"EVENT LOG  /  {_source.Entries.Count}/100  ·  {(_anchor is null ? "LIVE" : "HISTORY — scroll down for live")}";
58         for (var i = 0; i < 5; i++)
59         {
60             var message = i < entries.Length ? entries[i] : null;
61             _lines[i].Text = message is null ? " " : $"{TimeSpan.FromHours(8) + TimeSpan.FromMilliseconds(message.WorldMs):hh\\:mm\\:ss}  [{message.Kind.ToString().ToUpperInvariant()}]  {message.Text}";
62             _lines[i].TooltipText = message?.Text ?? "";
63             _lines[i].AddThemeColorOverride("font_color", message?.Kind switch
64             {
65                 MessageKind.Combat => new("efad8e"), MessageKind.Observation => new("e4d393"),
66                 MessageKind.Orders => new("a7c9ed"), MessageKind.Equipment => new("aacfb4"),
67                 MessageKind.Conversation => new("d5b7e4"), MessageKind.Warning => new("f3bf6d"), _ => new Color("c3ceca")
68             });
69         }
