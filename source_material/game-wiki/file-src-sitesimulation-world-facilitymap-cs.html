25         if (ids.Distinct().Count() != ids.Length) throw new ArgumentException("Duplicate fixture identity.");

28         var cells = new List<Hex>();

30         for (var r = -definition.Width; r <= definition.Height * 2; r++)

54     public IReadOnlyList<Hex> FindPath(Hex start, Hex end, IReadOnlySet<Hex>? occupied = null,
55         IReadOnlySet<Hex>? ignoredBlockers = null, bool approachBlockedGoal = false)
56     {
57         if (!Inside(end.World) || !approachBlockedGoal && (!Walkable(end, ignoredBlockers) || occupied?.Contains(end) == true)) return [];
58         var queue = new Queue<Hex>();
59         var previous = new Dictionary<Hex, Hex> { [start] = start };
60         var closest = start;
61         queue.Enqueue(start);
62         while (queue.TryDequeue(out var current))
63         {
64             if (current.Distance(end) < closest.Distance(end)) closest = current;
65             if (current == end)
66                 return Reconstruct(current);
67             for (var facing = 0; facing < 6; facing++)
68             {
69                 var next = current.Neighbor(facing);
70                 if (previous.ContainsKey(next) || !CanTraverse(current, next, ignoredBlockers) || occupied?.Contains(next) == true) continue;
71                 previous[next] = current;
72                 queue.Enqueue(next);
73             }
74         }
75         return approachBlockedGoal ? Reconstruct(closest) : [];
76 
77         IReadOnlyList<Hex> Reconstruct(Hex current)
78         {
79             var path = new List<Hex>();
80             while (current != start) { path.Add(current); current = previous[current]; }
81             path.Reverse();
82             return path;
83         }
