36     public static void Advance(SimulationWorld world)
37     {
38         var state = world.Operations;
39         if (state.Definition is not { } d) return;
40         var body = world.Anomalies.Bodies.Single(b => b.Id == d.SubjectId);
41         var report = ObservationService.Evaluate(world.Map, world.Actors, body.Position, world.LocalMs, body.ObserverProfile);
42         var atHome = body.Cell == d.Home;
43         if (!atHome) state.HadBreach = true;
44         var stable = atHome && state.Trolley.SubjectId is null && state.Trolley.Step is null
45             && report.Observers.Count(o => o.Kind == "human" && o.Detected && o.Eligible) >= 3
46             && d.AccessDoors.All(id => !world.Map.OpenDoors.Contains(id)) && world.Map.LightsOn;
47         state.StableMs = stable ? Math.Min(d.StabilizeMs, state.StableMs + SpatialSimulation.TickMs) : 0;
48         var status = state.StableMs >= d.StabilizeMs ? "Contained · procedure compliant"
49             : !atHome ? "Breach · outside receiving footprint"
50             : report.Observed ? "Emergency stabilized · procedure incomplete" : "Coverage lost · unresolved";
51         if (state.HadBreach && state.StableMs >= d.StabilizeMs) state.Completed = true;
52         if (status == state.PhysicalStatus) return;
53         state.PhysicalStatus = status;
54         world.Log("operation", d.SubjectId, "conditions", status);
55         var witness = world.Actors.FirstOrDefault(a => ObservationService.Inspect(world.Map, a, body.Position, world.LocalMs).Detected);
56         if (witness is null)
57         {
58             // Seeing the empty receiving footprint is evidence of absence, not of the subject's hidden location.
59             witness = world.Actors.FirstOrDefault(a => ObservationService.Inspect(world.Map, a, d.Home.World, world.LocalMs).Detected);
60             if (witness is not null && !atHome && (FacilityCommunications.Voice(world, witness, world.Player)
61                 || (state.TerminalPowered || OscpService.PhoneAvailable(world)) && FacilityCommunications.Radio(world, witness.Id)))
62                 state.Report(world, witness.Name, "Receiving footprint empty · subject location unconfirmed",
63                     source: FacilityCommunications.Voice(world, witness, world.Player) ? "voice" : "radio");
64             return;
65         }
66         if (FacilityCommunications.Voice(world, witness, world.Player) || (state.TerminalPowered || OscpService.PhoneAvailable(world)) && FacilityCommunications.Radio(world, witness.Id))
67             state.Report(world, witness.Name, $"Containment assessment: {status}", body.Position,
68                 source: FacilityCommunications.Voice(world, witness, world.Player) ? "voice" : "radio");
