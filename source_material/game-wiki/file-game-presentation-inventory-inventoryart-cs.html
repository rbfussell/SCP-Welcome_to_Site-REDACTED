14     private static void Load()
15     {
16         if (_loaded) return;
17         using var json = JsonDocument.Parse(Godot.FileAccess.GetFileAsString("res://Assets/inventory-objects/manifest.json"));
18         foreach (var asset in json.RootElement.GetProperty("assets").EnumerateArray())
19         {
20             var r = asset.GetProperty("region").EnumerateArray().Select(v => v.GetInt32()).ToArray();
21             Textures.Add(asset.GetProperty("id").GetString()!, new AtlasTexture
22             { Atlas = GD.Load<Texture2D>("res://Assets/inventory-objects/" + asset.GetProperty("file").GetString()), Region = new(r[0], r[1], r[2], r[3]) });
23         }
24         foreach (var binding in json.RootElement.GetProperty("attachmentIllustrations").EnumerateObject())
25         {
26             var profile = binding.Value.Deserialize<MountIllustration>(new JsonSerializerOptions { PropertyNameCaseInsensitive = true })!;
27             if (profile is null || !Textures.ContainsKey(binding.Name) || !float.IsFinite(profile.Height) || profile.Height is <= 0 or > 256
28                 || profile.Anchor is not { Length: 2 } || profile.Anchor.Any(v => !float.IsFinite(v) || v is < 0 or > 1)
29                 || profile.ProviderOrigin is { } origin && (origin.Length != 2 || origin.Any(v => !float.IsFinite(v) || v is < 0 or > 1))
30                 || !float.IsFinite(profile.RotationDegrees) || Math.Abs(profile.RotationDegrees) > 360)
31                 throw new InvalidOperationException("Invalid mounted illustration: " + binding.Name);
32             MountIllustrations.Add(binding.Name, profile);
33         }
34         _loaded = true;
35     }
36     private readonly record struct Layout(AtlasTexture Texture, Vector2 Center, Vector2 Size, bool Turn)
37     {
38         public Vector2 Project(Vector2 normalized)
39         {
40             var p = (normalized - Vector2.One * 0.5f) * Size;
41             return Center + (Turn ? new Vector2(-p.Y, p.X) : p);
42         }
43         public Vector2 Unproject(Vector2 screen)
44         {
45             var p = screen - Center;
46             return (Turn ? new Vector2(p.Y, -p.X) : p) / Size + Vector2.One * 0.5f;
47         }
48     }
49     private static Layout? GetLayout(string id, Rect2 bounds, bool rotated = false)
50     {
51         Load();
52         if (!Textures.TryGetValue(id, out var texture)) return null;
53         var turn = rotated ^ (id == "gum-pack");
54         var available = turn ? new Vector2(bounds.Size.Y, bounds.Size.X) : bounds.Size;
55         var scale = Math.Min(available.X / texture.GetWidth(), available.Y / texture.GetHeight());
56         return new(texture, bounds.GetCenter(), texture.GetSize() * scale, turn);
57     }

60     internal static bool Opaque(string id, Vector2 normalized)
61     {
62         if (normalized.X is < 0 or >= 1 || normalized.Y is < 0 or >= 1) return false;
63         Load(); var texture = Textures[id];
64         if (!Images.TryGetValue(id, out var image)) Images.Add(id, image = texture.Atlas.GetImage());
65         var pixel = texture.Region.Position + normalized * texture.Region.Size;
66         return image.GetPixel((int)pixel.X, (int)pixel.Y).A > 0.08f;
67     }
68     public static void Release()
69     {
70         foreach (var image in Images.Values) image.Dispose();
