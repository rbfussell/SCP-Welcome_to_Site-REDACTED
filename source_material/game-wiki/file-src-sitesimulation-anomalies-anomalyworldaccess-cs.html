21     public IReadOnlyList<RelocationCandidate> Candidates(string profile, RelocationQuery query)
22     {
23         if (profile != body.ObserverProfile || query.MinHexes < 1 || query.MaxHexes > body.MaxRelocationHexes || query.MaxHexes < query.MinHexes
24             || query.NearObserverKind is not ("human" or "ai")) return [];
25         var candidates = new List<RelocationCandidate>();
26         foreach (var actor in world.Actors.Where(a => a.ObserverKind == query.NearObserverKind
27             && (query.NearActorId is null || a.Id == query.NearActorId)).OrderBy(a => a.Id, StringComparer.Ordinal))
28         foreach (var cell in world.Map.Cells)
29         {
30             var distance = actor.Cell.Distance(cell);
31             if (distance < query.MinHexes || distance > query.MaxHexes || !Free(cell)) continue;
32             if (query.RequireClearLine && !world.Map.ClearSight(actor.Position, cell.World)) continue;
33             var behind = cell.World - actor.Position;
34             var direction = actor.Cell.Neighbor(actor.Facing).World - actor.Cell.World;
35             if (query.BehindOnly && behind.X * direction.X + behind.Y * direction.Y >= -0.01) continue;
36             if (ObservationService.Evaluate(world.Map, world.Actors, cell.World, LocalMs, profile).Observed) continue;
37             candidates.Add(new(cell, actor.Id, distance));
38         }
39         return candidates.OrderBy(p => p.Cell.Q).ThenBy(p => p.Cell.R)
40             .ThenBy(p => p.NearActorId, StringComparer.Ordinal).ToArray();
41     }

43     public bool TryRelocate(long expectedRevision, Hex destination, int pose, string profile, RelocationQuery query, out string reason)
44     {
45         reason = !world.LocalActive ? "local-held"
46             : profile != body.ObserverProfile ? "observer-profile-mismatch"
47             : expectedRevision != body.Revision ? "stale-body-revision"
48             : (pose < 0 || pose >= body.PoseCount) ? "invalid-pose"
49             : Observation(profile).Observed ? "origin-observed"
50             : !Free(destination) ? "destination-occupied-or-outside"
51             : ObservationService.Evaluate(world.Map, world.Actors, destination.World, LocalMs, profile).Observed ? "destination-observed"
52             : !Candidates(profile, query).Any(c => c.Cell == destination) ? "outside-relocation-policy" : "accepted";
53         if (reason != "accepted") return false;
54         var origin = body.Cell;
55         world.Map.DynamicBlockers.Remove(origin);
56         body.Cell = destination; body.Pose = pose; body.Revision++;
57         world.Map.DynamicBlockers.Add(destination);
58         ObservedTransportSystem.DetachRelocated(world, body.Id);
59         world.Log("anomaly-host", body.Id, "relocated", $"{origin.Q},{origin.R} -> {destination.Q},{destination.R}; pose {pose + 1}; silent");
60         return true;
61     }
