5 /// <summary>Map-local floor patches and ordered physical piles. Empty patches are resolved without mutating state.</summary>
6 public static class InventoryGround
7 {
8     public const string Prefix = "ground:";
9     public const string RegionId = "floor";
10     public const int Columns = 10, Rows = 10, CellMm = 50;
11     private const long MaxOrder = 1_000_000_000;
12     private static readonly StorageRegion Floor = new(RegionId, "Ground", Columns, Rows, CellMm, 1000, 100000,
13         AccessMs: 500, AccessAp: 2, Sealable: false, SecureOnClose: false);
14     public static string Id(Hex cell) => FormattableString.Invariant($"{Prefix}{cell.Q}:{cell.R}");
15     internal static InventoryRoot? FindRoot(InventoryState state, string id)
16     {
17         if (!state.Enabled || id is null || !id.StartsWith(Prefix, StringComparison.Ordinal)) return null;
18         var parts = id[Prefix.Length..].Split(':');
19         if (parts.Length != 2 || !int.TryParse(parts[0], NumberStyles.AllowLeadingSign, CultureInfo.InvariantCulture, out var q)
20             || !int.TryParse(parts[1], NumberStyles.AllowLeadingSign, CultureInfo.InvariantCulture, out var r)) return null;
21         var cell = new Hex(q, r);
22         return id == Id(cell) && state.GroundCells.Contains(cell)
23             ? new(id, FormattableString.Invariant($"Ground · {q}, {r}"), [Floor], MaxMassGrams: Floor.MaxMassGrams, GroundCell: cell) : null;
24     }
25     public static bool Overlap(InventoryState state, InventoryItem first, InventoryItem second)
26     {
27         var a = first.Placement; var b = second.Placement;
28         if (a.ParentId != b.ParentId || a.RegionId != b.RegionId) return false;
29         var region = state.Region(a.ParentId, a.RegionId);
30         var (aw, ah) = InventoryPlacement.Footprint(state.Definition(first.Id), region, a.Rotated);
31         var (bw, bh) = InventoryPlacement.Footprint(state.Definition(second.Id), region, b.Rotated);
32         return a.X < b.X + bw && a.X + aw > b.X && a.Y < b.Y + bh && a.Y + ah > b.Y;
33     }
34     /// <summary>Direct covering items; callers checking nested contents also traverse their ancestor locations.</summary>
35     public static IReadOnlyList<InventoryItem> Covering(InventoryState state, InventoryItem item)
36         => item.Placement.GroundOrder is not { } order ? [] : state.Children(item.Placement.ParentId, item.Placement.RegionId)
37             .Where(other => other.Placement.GroundOrder > order && Overlap(state, item, other))
38             .OrderByDescending(other => other.Placement.GroundOrder).ToArray();

40     internal static ItemPlacement Destination(InventoryState state, InventoryItem item, ItemPlacement target)
41     {
42         if (state.FindRoot(target.ParentId)?.GroundCell is null)
43         {
44             if (target.GroundOrder is not null) throw new ArgumentException("Pile order is only valid on the ground.");
45             return target;
46         }
47         var same = item.Placement with { GroundOrder = null } == target with { GroundOrder = null };
48         var order = same ? item.Placement.GroundOrder!.Value
49             : state.Children(target.ParentId, target.RegionId).Where(i => i.Id != item.Id).Select(i => i.Placement.GroundOrder ?? 0).DefaultIfEmpty(0).Max() + 1;
50         if (order is < 1 or > MaxOrder || target.GroundOrder is { } supplied && supplied != order)
51             throw new ArgumentException("Ground placement must go on top of the pile.");
52         return target with { GroundOrder = order };
53     }
54     internal static void Validate(InventoryState state)
55     {
56         foreach (var item in state.Items.Values)
57             if (state.FindRoot(item.Placement.ParentId)?.GroundCell is not null
58                 ? item.Placement.GroundOrder is not (>= 1 and <= MaxOrder)
59                 : item.Placement.GroundOrder is not null)
60                 throw new ArgumentException("Invalid or missing ground pile order.");
61         foreach (var pile in state.Items.Values.Where(i => i.Placement.GroundOrder is not null).GroupBy(i => i.Placement.ParentId))
62         {
63             var items = pile.ToArray(); var region = state.Region(pile.Key, RegionId);
64             if (items.Select(i => i.Placement.GroundOrder).Distinct().Count() != items.Length)
65                 throw new ArgumentException("Duplicate ground pile order.");
66             if (items.Sum(i => InventoryHierarchy.Mass(state, i.Id)) > region.MaxMassGrams)
67                 throw new ArgumentException("Ground patch exceeds its load limit.");
68             var height = new int[region.Rows * region.Columns];
69             foreach (var item in items)
70             {
71                 var p = item.Placement; var d = state.Definition(item.Id); var (w, h) = InventoryPlacement.Footprint(d, region, p.Rotated);
72                 for (var y = p.Y; y < p.Y + h; y++)
73                 for (var x = p.X; x < p.X + w; x++)
74                     if ((height[y * region.Columns + x] += d.ThicknessMm) > region.HeightMm)
75                         throw new ArgumentException("Ground pile is too high.");
76             }
77         }
78     }

85     public InventoryRequest GroundDropRequest(string id, bool rotated = false)
86     {
87         var root = InventoryGroundRoot;
88         if (root is null || !Inventory.Items.TryGetValue(id, out var item)) return new(InventoryActionKind.Move, id);
89         var region = root.Regions[0]; var (w, h) = InventoryPlacement.Footprint(Inventory.Definition(id), region, rotated);
90         // The explicit Put down shortcut prefers free floor, then the top of an existing pile.
91         for (var y = 0; y <= region.Rows - h; y++)
92         for (var x = 0; x <= region.Columns - w; x++)
93         {
94             var placed = item with { Placement = new(root.Id, region.Id, x, y, rotated) };
95             if (!Inventory.Children(root.Id, region.Id).Any(other => other.Id != id && InventoryGround.Overlap(Inventory, placed, other)))
96                 return new(InventoryActionKind.Move, id, placed.Placement);
97         }
98         return new(InventoryActionKind.Move, id, new(root.Id, region.Id, Rotated: rotated));
