16     public Scp650Module(AnomalyDefinition definition)
17     {
18         if (!ObservationService.ValidProfile(definition.ObserverProfile) || definition.Seed == 0)
19             throw new ArgumentException("SCP-650 requires a known observer profile and nonzero seed.");
20         InstanceId = definition.Id; _profile = definition.ObserverProfile;
21         _state = new() { RandomState = definition.Seed };
22     }
23     private uint NextRandom()
24     {
25         var x = _state.RandomState;
26         x ^= x << 13; x ^= x >> 17; x ^= x << 5;
27         return _state.RandomState = x;
28     }
29     public void Evaluate(IAnomalyWorld world, string cause)
30     {
31         var report = world.Observation(_profile);
32         var observers = report.EligibleIds;
33         if (!_state.LastObserverIds.SequenceEqual(observers))
34         {
35             world.Trace(observers.Length == 0 ? "observation-lost" : "observation-acquired",
36                 $"{cause}; eligible [{string.Join(",", observers)}]; profile {_profile}");
37             _state.LastObserverIds = observers;
38         }
39         if (_state.Relocations > 0)
40         foreach (var witness in report.Observers.Where(o => o.Detected && o.Kind == "human"))
41         {
42             if (_state.LastWitnessEncounter.GetValueOrDefault(witness.ActorId) == _state.Relocations) continue;
43             _state.LastWitnessEncounter[witness.ActorId] = _state.Relocations;
44             world.Witnessed(witness.ActorId, _state.Relocations);
45         }
46         if (report.Observed) { _state.LastReason = "observed; self-relocation suppressed"; return; }
47         if (world.LocalMs < _state.NextEvaluationMs) return;
48         // Complacency is explicitly authored for each person. Discovery alone never enables it.
49         // Threat poses appear close behind; relaxed poses use a wider, non-rear-only area.
50         var candidates = report.Observers.Where(o => o.Kind == "human")
51             .SelectMany(o => world.Candidates(_profile, Placement(o.ActorId)))
52             .OrderBy(c => c.Cell.Q).ThenBy(c => c.Cell.R).ThenBy(c => c.NearActorId, StringComparer.Ordinal).ToArray();
53         _state.LastCandidateCount = candidates.Length;
54         if (candidates.Length == 0)
55         {
56             _state.LastReason = "no valid unobserved destination";
57             _state.NextEvaluationMs = world.LocalMs + RetryMs;
58             world.Trace("retry-scheduled", $"{cause}; no candidate; retry at local {_state.NextEvaluationMs}ms");
59             return;
60         }
61         var beforeRandom = _state.RandomState;
62         // Choose a person, then a position: more available cells do not bias the target toward that person.
63         var targets = candidates.Select(c => c.NearActorId).Distinct().Order(StringComparer.Ordinal).ToArray();
64         var target = targets[(int)(NextRandom() % (uint)targets.Length)];
65         var positions = candidates.Where(c => c.NearActorId == target).ToArray();
66         var choice = positions[(int)(NextRandom() % (uint)positions.Length)];
67         var complacent = _state.ComplacentActorIds.Contains(choice.NearActorId);
68         var pose = complacent ? 3 + (int)(NextRandom() % 3u) : 1 + (int)(NextRandom() % 2u);
69         if (!world.TryRelocate(world.Body.Revision, choice.Cell, pose, _profile, Placement(choice.NearActorId), out var reason))
70         {
71             _state.RandomState = beforeRandom;
72             _state.LastReason = "commit rejected: " + reason;
73             _state.NextEvaluationMs = world.LocalMs + RetryMs;
74             world.Trace("relocation-rejected", reason);
75             return;
76         }
77         _state.Relocations++;
78         _state.LastReason = $"unobserved; relocated near {choice.NearActorId}";
79         _state.NextEvaluationMs = world.LocalMs + CooldownMs;
80         world.Trace("relocation-selected", $"{candidates.Length} valid candidates; {(complacent ? "relaxed" : "threat")} profile near {target}; {choice.DistanceHexes} hexes; encounter {_state.Relocations}; next evaluation {_state.NextEvaluationMs}ms");
81     }

86     public bool TrySetFlag(string flag, string actorId, string actorKind, bool enabled, out string reason)
87     {
88         reason = flag != "complacency" ? "unknown-module-flag" : actorKind != "human" ? "requires-human-target" : "accepted";
89         if (reason != "accepted") return false;
90         if (enabled) _state.ComplacentActorIds.Add(actorId); else _state.ComplacentActorIds.Remove(actorId);

101     public ModuleSnapshot Capture() => new(InstanceId, ModuleId, 2, JsonSerializer.Serialize(_state));
102     public void Restore(ModuleSnapshot snapshot, long localMs, IReadOnlyDictionary<string, string> actorKinds)
103     {
104         if (snapshot.Version != 2) throw new ArgumentException("SCP-650 save requires module schema 2 (explicit complacency); earlier prototype saves are not migrated.");
105         var state = JsonSerializer.Deserialize<Scp650State>(snapshot.Json) ?? throw new ArgumentException("Empty SCP state.");
106         if (snapshot.InstanceId != InstanceId || snapshot.ModuleId != ModuleId || snapshot.Version != 2 || state.RandomState == 0
107             || state.Relocations < 0 || state.LastCandidateCount < 0 || state.LastCandidateCount > 16384
108             || state.NextEvaluationMs < 0 || state.NextEvaluationMs > localMs + CooldownMs
109             || state.LastObserverIds is null || state.LastWitnessEncounter is null || state.LastReason is null || state.ComplacentActorIds is null
110             || state.ComplacentActorIds.Any(id => !actorKinds.TryGetValue(id, out var kind) || kind != "human")
111             || state.LastObserverIds.Any(id => !actorKinds.ContainsKey(id))
112             || state.LastWitnessEncounter.Any(p => !actorKinds.ContainsKey(p.Key) || p.Value < 1 || p.Value > state.Relocations))
113             throw new ArgumentException("Invalid SCP-650 continuation state.");
114         _state = state;
115     }
