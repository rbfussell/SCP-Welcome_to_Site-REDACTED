16         && Tokens(fit.FitsOver) && (fit.ExcludesUnderTags is null || Tokens(fit.ExcludesUnderTags));

18     /// <summary>Pure compatibility query. The caller supplies the chosen order; authored Layer never decides fit.</summary>
19     public static WearFitResult Evaluate(ItemDefinition outer, IReadOnlyList<ItemDefinition> beneath)
20     {
21         if (outer?.Wear?.Fit is not { } fit || !ProfileValid(fit))
22             return new(false, "Outer equipment has no valid authored fit profile.", 0, 0);
23         if (beneath is null || beneath.Count > 128 || beneath.Any(d => d?.Wear is null || d.Tags is null || !ProfileValid(d.Wear.Fit)))
24             return new(false, "Underlying equipment has no valid authored fit profile, or the fit query exceeds its safety bound.", 0, fit.MaxUnderBulk);
25         // Repeated definitions represent distinct physical garments and each contributes bulk.
26         var bulk = beneath.Sum(d => d.Wear!.Fit!.Bulk);
27         foreach (var inner in beneath.OrderBy(d => d.Id, StringComparer.Ordinal))
28         {
29             if (inner.Wear!.Outermost)
30                 return new(false, "Nothing can cover outermost " + inner.Name + "; remove it first.", bulk, fit.MaxUnderBulk);
31             if (!fit.FitsOver.Contains(inner.Wear.Fit!.Family, StringComparer.Ordinal))
32                 return new(false, outer.Name + " cannot fit over " + inner.Name + " (" + inner.Wear.Fit.Family + ").", bulk, fit.MaxUnderBulk);
33             if (fit.ExcludesUnderTags is { } excluded && inner.Tags.Any(excluded.Contains))
34                 return new(false, outer.Name + " excludes the fit traits of " + inner.Name + ".", bulk, fit.MaxUnderBulk);
35         }
36         return bulk > fit.MaxUnderBulk
37             ? new(false, outer.Name + $" has insufficient room: underlying bulk {bulk} exceeds clearance {fit.MaxUnderBulk}.", bulk, fit.MaxUnderBulk)
38             : new(true, "Fits within the authored clearance.", bulk, fit.MaxUnderBulk);
39     }

41     /// <summary>Validate compatibility beneath the actual saved dressing order at each declared body region.</summary>
42     internal static void ValidateWorn(InventoryState state, IReadOnlyList<InventoryItem> worn)
43     {
44         var cells = worn.ToDictionary(i => i.Id, i => InventoryWornStack.Cells(state, i).ToHashSet(), StringComparer.Ordinal);
45         foreach (var outer in worn)
46         {
47             var definition = state.Definition(outer.Id);
48             foreach (var cell in cells[outer.Id])
49             {
50                 var beneath = worn.Where(i => i.Id != outer.Id && InventoryWornOrder.Order(i) < InventoryWornOrder.Order(outer)
51                     && cells[i.Id].Contains(cell)).Select(i => state.Definition(i.Id)).ToArray();
52                 var result = Evaluate(definition, beneath);
53                 if (!result.Fits) throw new ArgumentException($"{cell.Surface} {cell.Cell.Q},{cell.Cell.R}: " + result.Reason);
