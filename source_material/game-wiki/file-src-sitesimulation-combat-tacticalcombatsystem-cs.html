6     public static bool Arm(SimulationWorld world, string opponentId, out string reason)
7     {
8         reason = "Finish the current drill or release the local hold first.";
9         if (world.Combat.Armed || !world.LocalActive) return false;
10         if (!world.Actors.Any(a => a.Id == opponentId && a != world.Player)) { reason = "Choose a staff training opponent."; return false; }
11         var state = world.Combat;
12         state.Roster.Clear(); state.Contacts.Clear();
13         foreach (var actor in world.Actors.OrderBy(a => a.Id, StringComparer.Ordinal))
14             state.Roster.Add(new(actor.Id, actor.Id == opponentId ? "opposition" : "friendly", 1 + (int)(state.Roll() % 20)));
15         state.Roster.Sort((a, b) => a.Initiative != b.Initiative ? b.Initiative.CompareTo(a.Initiative) : string.CompareOrdinal(a.ActorId, b.ActorId));
16         state.Armed = true; state.Round = 0; state.Cursor = 0;
17         reason = "Training armed: contact with the selected opponent starts local turns. No live weapons or injuries.";
18         PlayerFeedback.Add(world, MessageKind.Status, reason);
19         return true;
20     }
21     public static void Detect(SimulationWorld world)
22     {
23         var state = world.Combat;
24         if (!state.Armed || state.Active || !world.LocalActive) return;
25         if (!state.Roster.Any(c => c.Side == "opposition" && world.Player.Cell.Distance(Actor(world, c.ActorId).Cell) <= CombatState.EngagementHexes
26             && PlayerFeedback.Sees(world, Actor(world, c.ActorId)))) return;
27         // The facade evaluates due anomaly effects before reaching this scheduler. Entry is not a proximity shield.
28         InventoryActions.Cancel(world, "Combat began; reissue handling as a paid action");
29         foreach (var actor in world.Actors) MovementSystem.StopInPlace(actor);
30         ObservedTransportSystem.Interrupt(world);
31         state.Active = true; state.Round = 1;
32         BeginTurn(world);
33         PlayerFeedback.Add(world, MessageKind.Combat, "Training contact: local turns begin. The world calendar continues while you choose.");
34     }

43     public static bool CanAct(SimulationWorld world, string actorId, out string reason)
44     {
45         reason = world.ManualPause || world.LocalHold ? "Release the pause/hold first."
46             : !world.Combat.Active || world.Combat.CurrentActorId != actorId ? "Wait for your turn."
47             : world.Combat.Pending is not null ? "An action is still resolving."
48             : world.LocalMs < Actor(world, actorId).StartledUntilMs ? "Recovering from startle: end this turn to wait." : "accepted";
49         return reason == "accepted";
50     }
51     private static bool Commit(SimulationWorld world, string actorId, CombatActionKind kind, int cost, int duration,
52         out string reason, string? targetId = null, Point2? point = null, int steps = 0)
53     {
54         if (!CanAct(world, actorId, out reason)) return false;
55         var state = world.Combat; var participant = state.For(actorId)!;
56         if (participant.Ap < cost) { reason = $"Needs {cost} AP; {participant.Ap} remain."; return false; }
57         Replace(state, participant with { Ap = participant.Ap - cost });
58         state.Pending = new(state.NextActionId++, actorId, kind, cost, duration, duration, targetId, point, steps);
59         world.Log("combat", actorId, "action-committed", $"{kind}; {cost} AP; {duration}ms local");
60         reason = $"{kind} · {cost} AP";
61         return true;
62     }

65     public static bool Move(SimulationWorld world, string actorId, Hex destination, bool keepFacing, out string reason)
66     {
67         if (!CanAct(world, actorId, out reason)) return false;
68         var actor = Actor(world, actorId);
69         var path = actor == world.Player ? PlayerNavigation.PlanRoute(world, destination) : MovementSystem.PlanRoute(world, actor, destination);
70         if (path.Count == 0) { reason = "No closer reachable position."; return false; }
71         var duration = path.Count * SpatialSimulation.StepDurationMs;
72         if (actor.StoppedStep is { } stopped) duration -= path[0] == stopped.Next ? stopped.ProgressMs : SpatialSimulation.StepDurationMs - stopped.ProgressMs;
73         if (!Commit(world, actorId, CombatActionKind.Move, path.Count, duration, out reason, point: destination.World, steps: path.Count)) return false;
74         // No unpaid continuation or reroute: collision stops this purchased route and refunds untravelled steps.
75         MovementSystem.CommitRoute(actor, path, keepFacing);
76         return true;
77     }

88     {
89         if (!CanAct(world, actorId, out reason)) return false;

97     public static bool Fire(SimulationWorld world, string actorId, string targetId, out string reason)
98     {
99         if (!CanAct(world, actorId, out reason)) return false;
100         var state = world.Combat; var source = state.For(actorId)!; var target = state.For(targetId);
101         var actor = Actor(world, actorId);
102         if (target is null || target.Side == source.Side || target.Points == 0) { reason = "Choose an opposing training participant."; return false; }
103         var victim = Actor(world, targetId);
104         if (!source.WeaponReady || source.Ammo == 0) { reason = "Ready the training sidearm and load training rounds first."; return false; }
105         if (actor.Cell.Distance(victim.Cell) > CombatState.WeaponHexes || !ObservationService.Inspect(world.Map, actor, victim.Position, world.LocalMs).Detected)
106         { reason = "No visible target in training sidearm range."; return false; }
107         if (!Commit(world, actorId, CombatActionKind.Fire, CombatState.AttackCost, 600, out reason, targetId, victim.Position)) return false;
108         Replace(state, state.For(actorId)! with { Ammo = source.Ammo - 1 });
109         actor.Facing = ObservationService.FacingToward(actor.Position, victim.Position, actor.Facing);
110         world.Anomalies.Evaluate(world, "tactical-aim");
111         return true;

145     public static void ResolveTick(SimulationWorld world, Hex previousCell, Action? onResolvedPerception = null)
146     {
147         var state = world.Combat;
148         if (state.Pending is not { } pending) return;
149         var actor = Actor(world, pending.ActorId);
150         pending = pending with { RemainingMs = pending.RemainingMs - SpatialSimulation.TickMs,
151             CompletedSteps = pending.CompletedSteps + (actor.Cell != previousCell ? 1 : 0) };
152         if (pending.Kind == CombatActionKind.Move)
153             pending = pending with { RemainingMs = actor.Moving ? Math.Max(SpatialSimulation.TickMs, pending.RemainingMs) : 0 };
154         state.Pending = pending;
155         if (pending.RemainingMs > 0) return;
156         var c = state.For(actor.Id)!;
157         switch (pending.Kind)
158         {
159             case CombatActionKind.Move:
160                 var used = pending.CompletedSteps + (actor.StoppedStep is null ? 0 : 1);
161                 Replace(state, c with { Ap = Math.Min(CombatState.TurnAp, c.Ap + Math.Max(0, pending.CommittedSteps - used)) });
162                 break;
163             case CombatActionKind.ReadyWeapon:
164                 Replace(state, c with { WeaponReady = true }); PlayerFeedback.Equipment(world, actor, actor == world.Player ? "ready a training sidearm." : "readies a training sidearm."); break;
165             case CombatActionKind.Reload:
166                 var rounds = Math.Min(6 - c.Ammo, c.ReserveAmmo);
167                 Replace(state, c with { Ammo = c.Ammo + rounds, ReserveAmmo = c.ReserveAmmo - rounds });
168                 PlayerFeedback.Equipment(world, actor, actor == world.Player ? "reload training rounds." : "reloads training rounds."); break;
169             case CombatActionKind.Door:
170                 if (DoorInteraction.Validate(world, actor, pending.TargetId, out var door, out var doorReason))
171                 { DoorInteraction.Toggle(world, door!); PlayerFeedback.Equipment(world, actor, actor == world.Player ? "operate the door." : "operates the door."); }
172                 else if (PlayerFeedback.Sees(world, actor)) PlayerFeedback.Add(world, MessageKind.Warning, $"Door use interrupted: {doorReason}");
173                 break;
174             case CombatActionKind.Fire:
175                 var victim = Actor(world, pending.TargetId!); var target = state.For(victim.Id)!;
176                 var eligible = actor.Cell.Distance(victim.Cell) <= CombatState.WeaponHexes && ObservationService.Inspect(world.Map, actor, victim.Position, world.LocalMs).Detected;
177                 var roll = state.Roll(); var hit = eligible && roll % 100 < 75;
178                 var damage = hit ? Math.Min(2, target.Points) : 0;
179                 Replace(state, target with { Points = target.Points - damage });
180                 PlayerFeedback.Attack(world, actor, victim, hit, damage);
181                 break;
182         }
183         // Door/equipment outcomes can change observation before the next input hold exists.
184         world.Anomalies.Evaluate(world, "tactical-action-resolved");
185         onResolvedPerception?.Invoke();
186         PlayerFeedback.Sense(world);
187         world.Log("combat", actor.Id, "action-resolved", $"#{pending.Id} {pending.Kind}");
188         state.Pending = null;
189         if (pending.Kind == CombatActionKind.Wait) AdvanceTurn(world);
190         if (state.For(world.Player.Id)!.Points == 0 || !state.Roster.Any(p => p.Side == "opposition" && p.Points > 0)) FinishDrill(world, out _, true);
191     }
