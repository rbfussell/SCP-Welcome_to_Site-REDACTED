60         }

102     /// <summary>Pick only visible opposing sprites; transparent padding cannot intercept a ground click.</summary>
103     public ActorState? PickTrainingOpponent(Vector2 local)
104     {
105         foreach (var actor in Simulation.Actors.Where(a => a != Simulation.Player && Simulation.Combat.For(a.Id) is { Side: "opposition", Points: > 0 }
106             && Simulation.VisibleToPlayer(a.Position)).OrderByDescending(a => Screen(a.Position).Y))
107         {
108             foreach (var key in ActorLayers(actor).Reverse())
109             {
110                 var sprite = _sprites[key];
111                 var pixel = local - (Screen(actor.Position) - sprite.Anchor * sprite.Scale).Round();
112                 var size = sprite.Texture.GetSize() * sprite.Scale;
113                 if (pixel.X >= 0 && pixel.Y >= 0 && pixel.X < size.X && pixel.Y < size.Y)
114                 {
115                     using var image = sprite.Texture.GetImage();
116                     if (image.GetPixel((int)(pixel.X / sprite.Scale), (int)(pixel.Y / sprite.Scale)).A > 0.05f) return actor;
117                 }
118             }
119             if (Point2.Distance(actor.Position, ReferenceProjection.Unproject(new(local.X, local.Y))) < 0.3) return actor;
120         }
121         return null;
122     }

149     private void WallFace(WallDefinition wall, bool feather)
150     {
151         var start = Screen(wall.Start); var end = Screen(wall.End);
152         var height = RenderedWallHeight(wall);
153         var top = new Vector2(0, -height);
154         var light = LightAt(Screen(wall.Center)) * (wall.Axis == "u" ? Colors.White : new Color(0.79f, 0.84f, 0.88f));
155         Vector2[] face = [start, end, end + top, start + top];
156         var thickness = new Vector2(-3, -2);
157         Vector2[] cap = [start + top, end + top, end + top + thickness, start + top + thickness];
158         var capLight = new Color(ReducedWalls && !wall.Door ? "555b58" : "a5a99f") * light;
159         if (feather)
160         {
161             FeatheredQuad(wall.Id + ":face", face, light, _surfaceArt[wall.Door ? "door" : "wall"]);
162             FeatheredQuad(wall.Id + ":cap", cap, capLight);
163         }
164         else
165         {
166             DrawPolygon(face, new Color[] { light, light, light, light },
167                 new Vector2[] { new(0, 1), new(1, 1), new(1, 1 - height / WallHeight), new(0, 1 - height / WallHeight) }, _surfaceArt[wall.Door ? "door" : "wall"]);
168             DrawColoredPolygon(cap, capLight);
169         }
170         WallStroke(start, end, new Color("252b2b") * light, 2, feather);
171     }
