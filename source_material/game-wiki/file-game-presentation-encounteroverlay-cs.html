13     public Vector2 PlayerPoint { get; set; }
14     public IReadOnlyList<ContactBearing> Bearings { get; set; } = [];
15     public override void _Ready() => MouseFilter = MouseFilterEnum.Ignore;
16     public override void _Draw()
17     {
18         // Broad stepped edge shading stays outside the central readable area.
19         for (var i = 0; i < 14; i++)
20         {
21             var margin = i * 5f;
22             DrawRect(new Rect2(margin, margin, Size.X - margin * 2, Size.Y - margin * 2),
23                 new Color(0.015f, 0.02f, 0.025f, 0.035f), false, 10);
24         }
25         if (!ReducedEffects && Pulse > 0)
26             DrawRect(new Rect2(3, 3, Size.X - 6, Size.Y - 6), new Color(0.65f, 0.51f, 0.38f, (float)Pulse * 0.32f), false, 3);
27         foreach (var bearing in Bearings)
28         {
29             var direction = bearing.Direction.Normalized();
30             var tip = PlayerPoint + direction * 65;
31             if (!new Rect2(12, 12, Size.X - 24, Size.Y - 24).HasPoint(tip)) continue;
32             var side = new Vector2(-direction.Y, direction.X);
33             var color = bearing.Hostile ? new Color("df927b") : new Color("c3ac7b");
34             DrawColoredPolygon([tip, tip - direction * 11 + side * 5, tip - direction * 11 - side * 5], color);
35             var labelPoint = tip + side * 10 + new Vector2(0, -8);
36             DrawString(ThemeDB.FallbackFont, labelPoint, bearing.Label, HorizontalAlignment.Left, -1, 11, color);
37         }
38         if (Caption.Length == 0) return;
39         var font = ThemeDB.FallbackFont;
40         var width = font.GetStringSize(Caption, HorizontalAlignment.Left, -1, 19).X;
41         var p = new Vector2((Size.X - width) / 2, Size.Y - 45);
42         DrawRect(new Rect2(p.X - 20, p.Y - 25, width + 40, 40), new Color(0.025f, 0.035f, 0.035f, 0.92f));
43         DrawString(font, p, Caption, HorizontalAlignment.Left, -1, 19, new Color("ddc9af"));
44     }
