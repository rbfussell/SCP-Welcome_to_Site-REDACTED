9 /// <summary>Player knowledge and configurable movement interruption. Independent of SCP modules and fear.</summary>
10 public sealed class PlayerAwareness
11 {
12     private readonly Dictionary<string, PerceivedContact> _evidence = [];
13     private readonly Dictionary<string, long> _noticed = [];
14     public bool StopOnDirectSight { get; set; } = true;
15     public bool StopOnReports { get; set; } = true;
16     public IReadOnlyList<PerceivedContact> Evidence => _evidence.Values.OrderBy(e => e.SubjectId, StringComparer.Ordinal).ToArray();
17 
18     public bool Receive(PerceivedContact evidence)
19     {
20         if (string.IsNullOrWhiteSpace(evidence.SubjectId) || evidence.Encounter < 0 || evidence.LocalMs < 0
21             || string.IsNullOrWhiteSpace(evidence.ObserverId) || !double.IsFinite(evidence.Position.X) || !double.IsFinite(evidence.Position.Y)
22             || evidence.Source is not ("direct" or "voice" or "radio" or "sound")
23             || evidence.Classification is not ("anomaly" or "hostile" or "unknown")
24             || !double.IsFinite(evidence.UncertaintyRadius) || evidence.UncertaintyRadius < 0)
25             throw new ArgumentException("Invalid perceived contact evidence.");
26         if (_evidence.TryGetValue(evidence.SubjectId, out var previous)
27             && (evidence.LocalMs < previous.LocalMs || evidence.Encounter < previous.Encounter))
28             return false;
29         if (previous is null || evidence.LocalMs != previous.LocalMs || previous.Source != "direct" || evidence.Source == "direct")
30             _evidence[evidence.SubjectId] = evidence;
31         var enabled = evidence.Classification == "anomaly" && (evidence.Source == "direct" ? StopOnDirectSight : evidence.Source is "voice" or "radio" && StopOnReports);
32         var fresh = !_noticed.TryGetValue(evidence.SubjectId, out var encounter) || evidence.Encounter != encounter;
33         // A fresh known encounter is acknowledged even while stationary. Continuing sight is not another interruption.
34         if (enabled) _noticed[evidence.SubjectId] = evidence.Encounter;
35         return enabled && fresh;
36     }
37 
38     public AwarenessSnapshot Capture() => new(StopOnDirectSight, StopOnReports, Evidence.ToArray(),
39         _noticed.OrderBy(p => p.Key, StringComparer.Ordinal).Select(p => new NoticedAnomaly(p.Key, p.Value)).ToArray());
40 
41     public void Restore(AwarenessSnapshot snapshot, IReadOnlyDictionary<string, long> subjects, IReadOnlySet<string> actors, long localMs)
42     {
43         bool ValidSubject(PerceivedContact evidence) => subjects.TryGetValue(evidence.SubjectId, out var current)
44             ? evidence.Encounter >= 0 && evidence.Encounter <= current
45             : !string.IsNullOrWhiteSpace(evidence.SubjectId) && evidence.Source == "sound" && evidence.Classification == "unknown" && evidence.Encounter == 0;
46         if (snapshot.Evidence is null || snapshot.Noticed is null
47             || snapshot.Evidence.Any(e => e is null || !ValidSubject(e) || e.LocalMs < 0 || e.LocalMs > localMs
48                 || !double.IsFinite(e.Position.X) || !double.IsFinite(e.Position.Y)
49                 || e.Source is not ("direct" or "voice" or "radio" or "sound") || !actors.Contains(e.ObserverId)
50                 || e.Classification is not ("anomaly" or "hostile" or "unknown")
51                 || !double.IsFinite(e.UncertaintyRadius) || e.UncertaintyRadius < 0)
52             || snapshot.Noticed.Any(n => n is null || !subjects.TryGetValue(n.SubjectId, out var current) || n.Encounter < 0 || n.Encounter > current)
53             || snapshot.Evidence.Select(e => e.SubjectId).Distinct().Count() != snapshot.Evidence.Length
54             || snapshot.Noticed.Select(n => n.SubjectId).Distinct().Count() != snapshot.Noticed.Length)
55             throw new ArgumentException("Invalid player awareness state.");
56         _evidence.Clear(); _noticed.Clear();
57         foreach (var evidence in snapshot.Evidence) _evidence.Add(evidence.SubjectId, evidence);
58         foreach (var noticed in snapshot.Noticed) _noticed.Add(noticed.SubjectId, noticed.Encounter);
59         StopOnDirectSight = snapshot.StopOnDirectSight; StopOnReports = snapshot.StopOnReports;
60     }
