private static void AdvanceTurn(SimulationWorld world)
    {
        var state = world.Combat;
        do { state.Cursor++; if (state.Cursor == state.Roster.Count) { state.Cursor = 0; state.Round++; } }
        while (state.Roster[state.Cursor].Points == 0);
        BeginTurn(world);
    }
