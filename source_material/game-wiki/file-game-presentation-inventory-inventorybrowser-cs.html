30     internal bool Inspect(string id, out string reason)
31     {
32         var sim = Panel.Simulation; var state = sim.Inventory;
33         var root = state.Content!.Roots.FirstOrDefault(r => r.Id == id);
34         var item = state.DiagnosticItems.SingleOrDefault(i => i.Id == id);
35         reason = "Inspect its containing compartment first.";
36         if (root is not null)
37         {
38             if (!root.Regions.Any(r => sim.InventoryAccess(id, r.Id).Allowed)) { reason = "Approach the container with clear access."; return false; }
39             StartGear(id); return true;
40         }
41         if (item is null) return false;
42         var access = sim.InventoryInspectAccess(id);
43         if (!access.Allowed) { reason = access.Reason; return false; }
44         if (state.Content.Roots.Any(r => r.Id == item.Placement.ParentId && r.ActorId == sim.Player.Id))
45         { StartGear(id); return true; }
46         if (item.Placement.AttachmentPose is not null && CurrentGear is { } current)

48             var path = new List<string>(); var cursor = item;

50             {

52                 path.Add(cursor.Id);
53                 if (cursor.Placement.ParentId == current)
54                 {
55                     Panel.CancelDragPreview(); _context++; _opening = null;
56                     path.Reverse(); _gear.AddRange(path.Select(next => new Gear(next, Placement(next))));

59                 cursor = state.DiagnosticItems.Single(i => i.Id == cursor.Placement.ParentId);

71     internal bool OpenRegion(string owner, string region, out string reason)
72     {
73         var sim = Panel.Simulation;
74         reason = "Choose a compartment in the current gear or contents.";

76         var isGear = _gear.Any(g => g.Id == owner);
77         var item = sim.Inventory.DiagnosticItems.SingleOrDefault(i => i.Id == owner);
78         var nested = _contents.LastOrDefault() is { } parent && item?.Placement.ParentId == parent.Owner && item.Placement.RegionId == parent.Region;

81         // Validate parent access and coverage even if the lid is closed; the action service owns that check.

107     internal bool Dismiss(bool gear, bool entire)
108     {
109         if (gear ? _gear.Count == 0 : _contents.Count == 0) return false;
110         var removedGear = gear ? (entire ? _gear.ToArray() : [_gear[^1]]) : Array.Empty<Gear>();
111         var removedContents = !gear || _contents.Any(c => removedGear.Any(g => g.Id == c.Owner))
112             ? (entire || gear ? _contents.ToArray() : [_contents[^1]]) : Array.Empty<Interior>();
113         // Dismissing views has a physical default. A busy handler leaves the views in place with an explanation.
114         var interiors = gear && entire ? _visited.Concat(removedGear.SelectMany(g => g.Placement is null ? Array.Empty<Interior>()
115             : (Panel.Simulation.Inventory.Definition(g.Id).Regions ?? []).Where(r => r.Attachment is null).Select(r => new Interior(g.Id, r.Id, g.Placement)))) : removedContents;
116         var requests = interiors.Where(c => Placement(c.Owner) == c.Placement && Panel.Simulation.Inventory.IsOpen(c.Owner, c.Region))
117             .Select(c => new InventoryRequest(InventoryActionKind.Close, c.Owner, RegionId: c.Region))
118             .Concat(removedGear.Where(g => g.Placement is not null && Panel.Simulation.Inventory.Definition(g.Id).Wear?.OpenExposes is { Length: > 0 })
119                 .Select(g => new InventoryRequest(InventoryActionKind.CloseGarment, g.Id)))
120             .Where(r => Placement(r.ItemId) is not null).ToArray();
121         var automatic = InventoryClosures.OnDismissal(Panel.Simulation.Inventory, requests);
122         if (automatic.Length != 0 && !Panel.Simulation.SecureInventory(automatic, automatic: true)) { Panel.ShowReceipt(); return false; }
123         Panel.CancelDragPreview(); _context++; _opening = null;
124         if (gear) _gear.RemoveRange(entire ? 0 : _gear.Count - 1, entire ? _gear.Count : 1);
125         if (removedContents.Length != 0) _contents.RemoveRange(_contents.Count - removedContents.Length, removedContents.Length);
126         Build(); Panel.ShowReceipt(); return true;
127     }
