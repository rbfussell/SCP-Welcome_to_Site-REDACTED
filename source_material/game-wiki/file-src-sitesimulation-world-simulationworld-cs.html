25     public SimulationWorld(string mapJson, string? operationJson = null)
26     {
27         Map = FacilityMap.Load(mapJson);
28         Actors = Map.Definition.Actors.Select(a => new ActorState
29         {
30             Id = a.Id, Name = a.Name, Kind = a.Kind, ObserverKind = a.ObserverKind,
31             Cell = new(a.Q, a.R), Facing = a.Facing
32         }).ToList();
33         Anomalies = new AnomalySystem(Map.Definition.Anomalies ?? []);
34         Anomalies.RegisterBlockers(Map);
35         if (Actors.Any(a => !Map.Walkable(a.Cell))) throw new ArgumentException("Anomaly overlaps an actor spawn.");
36         Operations = new(operationJson);
37         Employment = new(Operations.Definition?.Employment);
38         Inventory = new(Operations.Definition?.InventoryProfile, Map);
39         OperationLifecycle.Initialize(this);
40         EmploymentSystem.Initialize(this);
41         InventoryContentValidation.Validate(this);
42     }
